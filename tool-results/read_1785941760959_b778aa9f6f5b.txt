    39→type LoanStep = 1 | 2 | 3 | 4 | 5 | 6
    40→
    41→// Format date to Korean
    42→function formatDate(dateStr: string): string {
    43→  const d = new Date(dateStr)
    44→  return `${d.getFullYear()}년 ${d.getMonth() + 1}월 ${d.getDate()}일`
    45→}
    46→
    47→// Add days to date
    48→function addDays(date: Date, days: number): Date {
    49→  const result = new Date(date)
    50→  result.setDate(result.getDate() + days)
    51→  return result
    52→}
    53→
    54→// Step progress bar
    55→function StepProgress({ current, total }: { current: number; total: number }) {
    56→  return (
    57→    <div className="mb-4">
    58→      <div className="flex items-center gap-1">
    59→        {Array.from({ length: total }).map((_, i) => (
    60→          <div
    61→            key={i}
    62→            className={cn(
    63→              'h-2 flex-1 rounded-full transition-colors',
    64→              i + 1 <= current ? 'bg-primary' : 'bg-muted'
    65→            )}
    66→          />
    67→        ))}
    68→      </div>
    69→      <p className="mt-1 text-center text-caption text-muted-foreground">
    70→        {current} / {total}단계
    71→      </p>
    72→    </div>
    73→  )
    74→}

    84→
    85→export default function CounterLoanView() {
    86→  const {
    87→    currentUser,
    88→    setView,
    89→    ttsEnabled,
    90→    isMissionMode,
    91→    completeMissionStep,
    92→    currentMissionStep,
    93→  } = useAppStore()
    94→
    95→  const [step, setStep] = useState<LoanStep>(1)
    96→  const [cardPresented, setCardPresented] = useState(false)
    97→  const [books, setBooks] = useState<BookItem[]>([])
    98→  const [activeLoans, setActiveLoans] = useState<LoanItem[]>([])
    99→  const [selectedBooks, setSelectedBooks] = useState<BookItem[]>([])
   100→  const [loading, setLoading] = useState(false)
   101→  const [confirmOpen, setConfirmOpen] = useState(false)
   102→  const [loansCreated, setLoansCreated] = useState(false)
   103→  const [loanResults, setLoanResults] = useState<{ book: BookItem; dueDate: string }[]>([])
   104→  const [librarianMessage, setLibrarianMessage] = useState('')
   105→  const [cardAnimation, setCardAnimation] = useState(false)
   106→
   107→  // Fetch available books and active loans
   108→  const refreshData = async () => {
   109→    if (!currentUser) return
   110→    try {
   111→      const [booksRes, loansRes] = await Promise.all([
   112→        fetch('/api/books'),
   113→        fetch(`/api/loans?userId=${currentUser.id}`),
   114→      ])
   115→      if (booksRes.ok) {
   116→        const booksData = await booksRes.json()
   117→        setBooks(booksData.filter((b: BookItem) => b.availableCopies > 0))
   118→      }
   119→      if (loansRes.ok) {
   120→        const loansData = await loansRes.json()
   121→        const active = loansData.filter((l: LoanItem) => l.status === 'active')
   122→        setActiveLoans(active)
   123→      }
   124→    } catch {
   125→      // ignore
   126→    }
   127→  }
   128→
   129→  useEffect(() => {
   130→    let cancelled = false
   131→    async function loadData() {
   132→      if (!currentUser) return
   133→      try {
   134→        const [booksRes, loansRes] = await Promise.all([
   135→          fetch('/api/books'),
   136→          fetch(`/api/loans?userId=${currentUser.id}`),
   137→        ])
   138→        if (cancelled) return
   139→        if (booksRes.ok) {
   140→          const booksData = await booksRes.json()
   141→          setBooks(booksData.filter((b: BookItem) => b.availableCopies > 0))
   142→        }
   143→        if (loansRes.ok) {
   144→          const loansData = await loansRes.json()
   145→          const active = loansData.filter((l: LoanItem) => l.status === 'active')
   146→          setActiveLoans(active)
   147→        }
   148→      } catch {
   149→        // ignore
   150→      }
   151→    }
   152→    loadData()
   153→    return () => { cancelled = true }
   154→  }, [currentUser])
   155→
   156→  // TTS on step change
   157→  useEffect(() => {
   158→    if (!ttsEnabled) return
   159→    const messages: Record<LoanStep, string> = {
   160→      1: '사서님께 도서증을 제시해주세요.',
   161→      2: '빌리고 싶은 책을 선택해주세요. 최대 5권까지 빌릴 수 있습니다.',
   162→      3: '대여 권수를 확인해주세요.',
   163→      4: '연체된 도서가 있는지 확인해주세요.',
   164→      5: '대여할 책과 반납일을 확인해주세요.',
   165→      6: '대여가 완료되었습니다.',
   166→    }
   167→    speak(messages[step])
   168→  }, [step, ttsEnabled])
   169→
   170→  // Pre-requisite check
   171→  if (!currentUser) {
   172→    return (
   173→      <div className="flex min-h-screen flex-col bg-background">
   174→        <TopBar title="창구 대여" />
   175→        <main className="mx-auto flex w-full max-w-lg flex-1 flex-col items-center justify-center gap-6 px-4">
   176→          <CreditCard className="h-16 w-16 text-muted-foreground/50" />
   177→          <h2 className="text-title text-foreground">회원가입이 필요합니다</h2>
   178→          <p className="text-body text-muted-foreground text-center">
   179→            먼저 회원가입을 해주세요
   180→          </p>
   181→          <button
   182→            onClick={() => setView('registration')}
   183→            className="btn-senior bg-primary text-primary-foreground rounded-xl"
   184→          >
   185→            <UserCheck className="h-6 w-6" />
   186→            회원가입하기
   187→          </button>
   188→        </main>
   189→      </div>
   190→    )
   191→  }
   192→
   193→  // Toggle book selection
   194→  const toggleBook = (book: BookItem) => {
   195→    setSelectedBooks((prev) => {
   196→      const exists = prev.find((b) => b.id === book.id)
   197→      if (exists) {
   198→        return prev.filter((b) => b.id !== book.id)
   199→      }
   200→      if (prev.length >= MAX_LOAN_COUNT) {
   201→        toast.warning(`최대 ${MAX_LOAN_COUNT}권까지 선택할 수 있습니다.`)
   202→        return prev
   203→      }
   204→      return [...prev, book]
   205→    })
   206→  }
   207→
   208→  // Step 1: Present card
   209→  const handlePresentCard = () => {
   210→    setCardAnimation(true)
   211→    setLibrarianMessage('네, 확인했습니다')
   212→    setTimeout(() => {
   213→      setCardPresented(true)
   214→      if (ttsEnabled) speak('네, 확인했습니다. 도서증을 확인했습니다.')
   215→      if (isMissionMode) completeMissionStep()
   216→    }, 1500)
   217→  }
   218→
   219→  // Go to next step after selection
   220→  const handleNextFromSelection = () => {
   221→    if (selectedBooks.length === 0) {
   222→      toast.warning('최소 1권 이상 선택해주세요.')
   223→      return
   224→    }
   225→
   226→    // Check if already at max loans
   227→    const totalLoans = activeLoans.length + selectedBooks.length
   228→    if (totalLoans > MAX_LOAN_COUNT) {
   229→      setStep(3)
   230→      return
   231→    }
   232→
   233→    // Check for overdue loans
   234→    const today = new Date()
   235→    const hasOverdue = activeLoans.some((loan) => new Date(loan.dueDate) < today)
   236→    if (hasOverdue) {
   237→      setStep(4)
   238→      return
   239→    }
   240→
   241→    setStep(5)
   242→    if (isMissionMode) completeMissionStep()
   243→  }
   244→
   245→  // Step 5: Confirm loan
   246→  const handleConfirmLoan = async () => {
   247→    setConfirmOpen(false)
   248→    setLoading(true)
   249→
   250→    const today = new Date()
   251→    const results: { book: BookItem; dueDate: string }[] = []
   252→
   253→    for (const book of selectedBooks) {
   254→      try {
   255→        const res = await fetch('/api/loans', {
   256→          method: 'POST',
   257→          headers: { 'Content-Type': 'application/json' },
   258→          body: JSON.stringify({
   259→            userId: currentUser.id,
   260→            bookId: book.id,
   261→            method: 'counter',
   262→          }),
   263→        })
   264→        if (res.ok) {
   265→          const dueDate = addDays(today, LOAN_PERIOD_DAYS)
   266→          results.push({
   267→            book,
   268→            dueDate: dueDate.toISOString().split('T')[0],
   269→          })
   270→        } else {
   271→          const data = await res.json()
   272→          toast.error(data.error || `${book.title} 대출에 실패했습니다.`)
   273→        }
   274→      } catch {
   275→        toast.error('네트워크 오류가 발생했습니다.')
   276→      }
   277→    }
   278→
   279→    setLoanResults(results)
   280→    setLoansCreated(true)
   281→    setStep(6)
   282→    setLoading(false)
   283→
   284→    if (results.length > 0) {
   285→      if (ttsEnabled) {
   286→        speak(`대여가 완료되었습니다. ${results.length}권을 빌렸습니다.`)
   287→      }
   288→      if (isMissionMode) completeMissionStep()
   289→    }
   290→
   291→    // Refresh active loans
   292→    refreshData()
   293→  }
   294→
   295→  return (
   296→    <div className="flex min-h-screen flex-col bg-background">
   297→      <TopBar
   298→        title="창구 대여"
   299→        helpText="창구 대여 화면입니다. 사서님께 도서증을 보여주고 책을 빌리는 연습을 합니다."
   300→      />
   301→
   302→      <main className="mx-auto w-full max-w-lg flex-1 px-4 py-6">
   303→        <StepProgress current={step} total={6} />
   304→
   305→        <AnimatePresence mode="wait">
   306→          {/* Step 1: Present Card */}
   307→          {step === 1 && (
   308→            <motion.div
   309→              key="step1"
   310→              initial={{ opacity: 0, x: 50 }}
   311→              animate={{ opacity: 1, x: 0 }}
   312→              exit={{ opacity: 0, x: -50 }}
   313→              className="flex flex-col gap-6"
   314→            >
   315→              <h2 className="text-title text-foreground text-center">
   316→                도서증 제시
   317→              </h2>
   318→
   319→              {/* Counter desk scene */}
   320→              <div className="card-senior flex flex-col items-center gap-4">
   321→                {/* Desk illustration */}
   322→                <div className="w-full rounded-xl bg-amber-100/60 p-6">
   323→                  <div className="flex items-center gap-4">
   324→                    {/* Librarian avatar */}
   325→                    <div className="flex h-16 w-16 shrink-0 items-center justify-center rounded-full bg-emerald-500">
   326→                      <span className="text-heading text-white font-bold">사</span>
   327→                    </div>
   328→                    <div className="flex-1">
   329→                      <p className="text-body font-bold text-foreground">사서님</p>
   330→                      <p className="text-caption text-muted-foreground">
   331→                        도서증을 보여주세요
   332→                      </p>
   333→                    </div>
   334→                    <Library className="h-8 w-8 text-emerald-600" />
   335→                  </div>
   336→                </div>
   337→
   338→                {/* User's card */}
   339→                <motion.div
   340→                  animate={cardAnimation ? { y: [0, -20, -60], opacity: [1, 1, 0] } : {}}
   341→                  transition={{ duration: 1.2 }}
   342→                  className={cn(
   343→                    'w-full rounded-xl border-2 bg-gradient-to-br from-emerald-600 to-teal-500 p-4 text-white shadow-lg',
   344→                    cardAnimation && 'opacity-0'
   345→                  )}
   346→                >
   347→                  <div className="flex items-center justify-between">
   348→                    <div>
   349→                      <p className="text-caption text-white/70">스마트 도서관</p>
   350→                      <p className="text-heading font-bold">{currentUser.name}</p>
   351→                      <p className="text-body mt-1 font-mono">{currentUser.cardNumber}</p>
   352→                    </div>
   353→                    <CreditCard className="h-10 w-10 text-white/60" />
   354→                  </div>
   355→                </motion.div>
   356→
   357→                {/* Librarian message */}
   358→                {librarianMessage && (
   359→                  <motion.div
   360→                    initial={{ opacity: 0, scale: 0.9 }}
   361→                    animate={{ opacity: 1, scale: 1 }}
   362→                    className="w-full rounded-xl bg-emerald-50 border-2 border-emerald-200 p-4"
   363→                  >
   364→                    <div className="flex items-center gap-3">
   365→                      <CheckCircle className="h-6 w-6 text-emerald-600" />
   366→                      <p className="text-body font-bold text-emerald-800">
   367→                        {librarianMessage}
   368→                      </p>
   369→                    </div>
   370→                  </motion.div>
   371→                )}
   372→
   373→                {!cardPresented && (
   374→                  <button
   375→                    onClick={handlePresentCard}
   376→                    className="btn-senior w-full bg-primary text-primary-foreground rounded-xl"
   377→                  >
   378→                    <CreditCard className="h-6 w-6" />
   379→                    도서증을 제시합니다
   380→                  </button>
   381→                )}
   382→
   383→                {cardPresented && (
   384→                  <button
   385→                    onClick={() => {
   386→                      setStep(2)
   387→                      if (isMissionMode) completeMissionStep()
   388→                    }}
   389→                    className="btn-senior w-full bg-primary text-primary-foreground rounded-xl"
   390→                  >
   391→                    <ChevronRight className="h-6 w-6" />
   392→                    다음 단계로
   393→                  </button>
   394→                )}
   395→              </div>
   396→            </motion.div>
   397→          )}
   398→
   399→          {/* Step 2: Select books */}
   400→          {step === 2 && (
   401→            <motion.div
   402→              key="step2"
   403→              initial={{ opacity: 0, x: 50 }}
   404→              animate={{ opacity: 1, x: 0 }}
   405→              exit={{ opacity: 0, x: -50 }}
   406→              className="flex flex-col gap-6"
   407→            >
   408→              <h2 className="text-title text-foreground text-center">
   409→                빌릴 책 선택
   410→              </h2>
   411→
   412→              {/* Current active loans count */}
   413→              <div className="flex items-center gap-4">
   414→                <div className="card-senior flex flex-1 items-center gap-3">
   415→                  <BookOpen className="h-6 w-6 text-primary" />
   416→                  <div>
   417→                    <p className="text-caption text-muted-foreground">현재 대출</p>
   418→                    <p className="text-body font-bold text-foreground">{activeLoans.length}권</p>
   419→                  </div>
   420→                </div>
   421→                <div className="card-senior flex flex-1 items-center gap-3">
   422→                  <BookOpen className="h-6 w-6 text-emerald-500" />
   423→                  <div>
   424→                    <p className="text-caption text-muted-foreground">대출 가능</p>
   425→                    <p className="text-body font-bold text-foreground">
   426→                      {Math.max(0, MAX_LOAN_COUNT - activeLoans.length)}권
   427→                    </p>
   428→                  </div>
   429→                </div>
   430→              </div>
   431→
   432→              {/* Selected books summary */}
   433→              {selectedBooks.length > 0 && (
   434→                <div className="rounded-xl bg-primary/5 border-2 border-primary/20 p-4">
   435→                  <p className="text-body font-bold text-primary mb-2">
   436→                    선택한 책: {selectedBooks.length}권
   437→                  </p>
   438→                  <div className="flex flex-wrap gap-2">
   439→                    {selectedBooks.map((b) => (
   440→                      <Badge key={b.id} variant="secondary" className="text-caption py-1.5 px-3">
   441→                        {b.title}
   442→                        <button
   443→                          onClick={() => toggleBook(b)}
   444→                          className="ml-2 text-muted-foreground hover:text-foreground"
   445→                        >
   446→                          <X className="h-4 w-4" />
   447→                        </button>
   448→                      </Badge>
   449→                    ))}
   450→                  </div>
   451→                </div>
   452→              )}
   453→
   454→              {/* Book list */}
   455→              <div className="max-h-80 overflow-y-auto custom-scrollbar flex flex-col gap-3">
   456→                {books.map((book, index) => {
   457→                  const isSelected = selectedBooks.some((b) => b.id === book.id)
   458→                  const colors = CATEGORY_COLORS[book.category] || { bg: 'bg-slate-100', icon: 'text-slate-600' }
   459→
   460→                  return (
   461→                    <motion.div
   462→                      key={book.id}
   463→                      initial={{ opacity: 0, y: 10 }}
   464→                      animate={{ opacity: 1, y: 0 }}
   465→                      transition={{ delay: index * 0.03 }}
   466→                      onClick={() => toggleBook(book)}
   467→                      className={cn(
   468→                        'card-senior cursor-pointer transition-all',
   469→                        isSelected && 'border-primary bg-primary/5'
   470→                      )}
   471→                    >
   472→                      <div className="flex items-center gap-3">
   473→                        {/* Checkbox */}
   474→                        <div
   475→                          className="flex h-12 w-12 shrink-0 items-center justify-center rounded-lg border-2 transition-colors"
   476→                          style={{
   477→                            borderColor: isSelected ? 'var(--primary)' : 'var(--border)',
   478→                            background: isSelected ? 'var(--primary)' : 'var(--card)',
   479→                          }}
   480→                        >
   481→                          {isSelected && (
   482→                            <CheckCircle className="h-8 w-8 text-primary-foreground" />
   483→                          )}
   484→                        </div>
   485→
   486→                        {/* Book cover */}
   487→                        <div className={cn('flex h-14 w-10 shrink-0 items-center justify-center rounded-lg', colors.bg)}>
   488→                          <BookOpen className={cn('h-6 w-6', colors.icon)} />
   489→                        </div>
   490→
   491→                        {/* Info */}
   492→                        <div className="flex-1 min-w-0">
   493→                          <h4 className="text-body font-bold text-foreground truncate">{book.title}</h4>
   494→                          <p className="text-caption text-muted-foreground truncate">{book.author}</p>
   495→                        </div>
   496→
   497→                        {/* Available copies */}
   498→                        <Badge className="bg-emerald-100 text-emerald-800" variant="secondary">
   499→                          {book.availableCopies}권
   500→                        </Badge>
   501→                      </div>
   502→                    </motion.div>
   503→                  )
   504→                })}
   505→              </div>
   506→
   507→              {/* Next button */}
   508→              <div className="flex gap-3">
   509→                <button
   510→                  onClick={() => setStep(1)}
   511→                  className="btn-senior bg-secondary text-secondary-foreground rounded-xl"
   512→                >
   513→                  이전
   514→                </button>
   515→                <button
   516→                  onClick={handleNextFromSelection}
   517→                  className="btn-senior flex-1 bg-primary text-primary-foreground rounded-xl"
   518→                >
   519→                  <ArrowRight className="h-5 w-5" />
   520→                  다음 ({selectedBooks.length}권)
   521→                </button>
   522→              </div>
   523→            </motion.div>
   524→          )}
   525→
   526→          {/* Step 3: Loan limit exceeded */}
   527→          {step === 3 && (
   528→            <motion.div
   529→              key="step3"
   530→              initial={{ opacity: 0, x: 50 }}
   531→              animate={{ opacity: 1, x: 0 }}
   532→              exit={{ opacity: 0, x: -50 }}
   533→              className="flex flex-col items-center gap-6"
   534→            >
   535→              <motion.div
   536→                initial={{ scale: 0 }}
   537→                animate={{ scale: 1 }}
   538→                transition={{ type: 'spring' }}
   539→              >
   540→                <AlertTriangle className="h-20 w-20 text-amber-500" />
   541→              </motion.div>
   542→
   543→              <div className="text-center">
   544→                <h2 className="text-title text-foreground">
   545→                  대여 권수를 초과했습니다!
   546→                </h2>
   547→                <p className="text-body text-muted-foreground mt-2">
   548→                  최대 {MAX_LOAN_COUNT}권까지 대출할 수 있습니다.
