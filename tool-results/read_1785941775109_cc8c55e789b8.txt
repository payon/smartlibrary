    30→// Format date to Korean
    31→function formatDate(dateStr: string): string {
    32→  const d = new Date(dateStr)
    33→  return `${d.getFullYear()}년 ${d.getMonth() + 1}월 ${d.getDate()}일`
    34→}
    35→
    36→// Get days remaining (negative = overdue)
    37→function getDaysRemaining(dueDate: string): number {
    38→  const today = new Date()
    39→  today.setHours(0, 0, 0, 0)
    40→  const due = new Date(dueDate)
    41→  due.setHours(0, 0, 0, 0)
    42→  return Math.ceil((due.getTime() - today.getTime()) / (1000 * 60 * 60 * 24))
    43→}
    44→
    45→// Get color class based on days remaining
    46→function getDaysColor(days: number): string {
    47→  if (days < 0) return 'bg-rose-100 text-rose-800'
    48→  if (days <= 7) return 'bg-amber-100 text-amber-800'
    49→  return 'bg-emerald-100 text-emerald-800'
    50→}
    51→
    52→function getDaysLabel(days: number): string {
    53→  if (days < 0) return `${Math.abs(days)}일 연체`
    54→  if (days === 0) return '오늘 반납'
    55→  return `${days}일 남음`
    56→}
    57→
    58→// Step progress bar
    59→function StepProgress({ current, total }: { current: number; total: number }) {
    60→  return (
    61→    <div className="mb-4">
    62→      <div className="flex items-center gap-1">
    63→        {Array.from({ length: total }).map((_, i) => (
    64→          <div
    65→            key={i}
    66→            className={cn(
    67→              'h-2 flex-1 rounded-full transition-colors',
    68→              i + 1 <= current ? 'bg-primary' : 'bg-muted'
    69→            )}
    70→          />
    71→        ))}
    72→      </div>
    73→      <p className="mt-1 text-center text-caption text-muted-foreground">
    74→        {current} / {total}단계
    75→      </p>
    76→    </div>
    77→  )
    78→}
    79→

    89→            width: i % 3 === 0 ? '2px' : '1px',
    90→          }}
    91→        />
    92→      ))}
    93→    </div>
    94→  )
    95→}

    99→  book: LoanItem
   100→  index: number
   101→  total: number
   102→  scanning: boolean
   103→  scanned: boolean

   106→    <motion.div

   186→export default function CounterReturnView() {
   187→  const {
   188→    currentUser,
   189→    setView,
   190→    ttsEnabled,
   191→    isMissionMode,
   192→    completeMissionStep,
   193→  } = useAppStore()
   194→
   195→  const [step, setStep] = useState<ReturnStep>(1)
   196→  const [activeLoans, setActiveLoans] = useState<LoanItem[]>([])
   197→  const [selectedLoans, setSelectedLoans] = useState<LoanItem[]>([])
   198→  const [loading, setLoading] = useState(true)
   199→  const [scanningIndex, setScanningIndex] = useState(-1)
   200→  const [scannedIndices, setScannedIndices] = useState<Set<number>>(new Set())
   201→  const [totalFee, setTotalFee] = useState(0)
   202→  const [overdueItems, setOverdueItems] = useState<{ book: LoanItem; days: number; fee: number }[]>([])
   203→  const [returnCompleted, setReturnCompleted] = useState(false)
   204→  const [hasOverdue, setHasOverdue] = useState(false)
   205→
   206→  // Fetch active loans
   207→  const fetchLoans = useCallback(async () => {
   208→    if (!currentUser) return
   209→    setLoading(true)
   210→    try {
   211→      const res = await fetch(`/api/loans?userId=${currentUser.id}`)
   212→      if (res.ok) {
   213→        const data = await res.json()
   214→        const active = data.filter((l: LoanItem) => l.status === 'active')
   215→        setActiveLoans(active)
   216→      }
   217→    } catch {
   218→      // ignore
   219→    } finally {
   220→      setLoading(false)
   221→    }
   222→  }, [currentUser])
   223→
   224→  useEffect(() => {
   225→    fetchLoans()
   226→  }, [fetchLoans])
   227→
   228→  // TTS on step change
   229→  useEffect(() => {
   230→    if (!ttsEnabled) return
   231→    const messages: Record<ReturnStep, string> = {
   232→      1: '반납할 책을 선택해주세요.',
   233→      2: '책 바코드를 스캔하고 있습니다.',
   234→      3: '연체료를 확인해주세요.',
   235→      4: '반납이 완료되었습니다.',
   236→    }
   237→    speak(messages[step])
   238→  }, [step, ttsEnabled])
   239→
   240→  // Toggle loan selection
   241→  const toggleLoan = (loan: LoanItem) => {
   242→    setSelectedLoans((prev) => {
   243→      const exists = prev.find((l) => l.id === loan.id)
   244→      if (exists) {
   245→        return prev.filter((l) => l.id !== loan.id)
   246→      }
   247→      return [...prev, loan]
   248→    })
   249→  }
   250→
   251→  // Run scan simulation
   252→  const runScanSimulation = () => {
   253→    if (selectedLoans.length === 0) return
   254→    setStep(2)
   255→    setScanningIndex(0)
   256→    setScannedIndices(new Set())
   257→
   258→    let currentIndex = 0
   259→    const interval = setInterval(() => {
   260→      // Mark current as scanned
   261→      setScannedIndices((prev) => {
   262→        const next = new Set(prev)
   263→        next.add(currentIndex)
   264→        return next
   265→      })
   266→
   267→      if (ttsEnabled && currentIndex === selectedLoans.length - 1) {
   268→        speak('삑! 모든 책 스캔이 완료되었습니다.')
   269→      }
   270→
   271→      currentIndex++
   272→      if (currentIndex >= selectedLoans.length) {
   273→        setScanningIndex(-1)
   274→        clearInterval(interval)
   275→
   276→        // Calculate overdue fees after a brief delay
   277→        setTimeout(() => {
   278→          let total = 0
   279→          const items: { book: LoanItem; days: number; fee: number }[] = []
   280→          for (const loan of selectedLoans) {
   281→            const days = getDaysRemaining(loan.dueDate)
   282→            if (days < 0) {
   283→              const fee = Math.abs(days) * OVERDUE_FEE_PER_DAY
   284→              items.push({ book: loan, days: Math.abs(days), fee })
   285→              total += fee
   286→            }
   287→          }
   288→          setOverdueItems(items)
   289→          setTotalFee(total)
   290→          setHasOverdue(total > 0)
   291→
   292→          if (total > 0) {
   293→            setStep(3)
   294→          } else {
   295→            setStep(3) // still go to step 3 to confirm
   296→          }
   297→
   298→          if (isMissionMode) completeMissionStep()
   299→        }, 1000)
   300→      } else {
   301→        setScanningIndex(currentIndex)
   302→      }
   303→    }, 1500)
   304→  }
   305→
   306→  // Process return
   307→  const processReturn = async () => {
   308→    setLoading(true)
   309→
   310→    let anyOverdue = false
   311→    let calculatedTotalFee = 0
   312→    const items: { book: LoanItem; days: number; fee: number }[] = []
   313→
   314→    for (const loan of selectedLoans) {
   315→      try {
   316→        const res = await fetch(`/api/loans/${loan.id}/return`, {
   317→          method: 'POST',
   318→        })
   319→        if (res.ok) {
   320→          const data = await res.json()
   321→          if (data.overdueDays > 0) {
   322→            anyOverdue = true
   323→            const fee = data.overdueDays * OVERDUE_FEE_PER_DAY
   324→            calculatedTotalFee += fee
   325→            items.push({ book: loan, days: data.overdueDays, fee })
   326→          }
   327→        }
   328→      } catch {
   329→        toast.error(`${loan.book?.title} 반납에 실패했습니다.`)
   330→      }
   331→    }
   332→
   333→    setOverdueItems(items)
   334→    setTotalFee(calculatedTotalFee)
   335→    setHasOverdue(anyOverdue)
   336→    setReturnCompleted(true)
   337→    setStep(4)
   338→    setLoading(false)
   339→
   340→    if (ttsEnabled) {
   341→      if (anyOverdue) {
   342→        speak(`반납이 완료되었습니다. 연체료는 ${calculatedTotalFee.toLocaleString()}원입니다.`)
   343→      } else {
   344→        speak('반납이 완료되었습니다. 감사합니다!')
   345→      }
   346→    }
   347→
   348→    if (isMissionMode) completeMissionStep()
   349→
   350→    // Refresh loans
   351→    fetchLoans()
   352→  }
   353→
   354→  // Pre-requisite check
   355→  if (!currentUser) {
   356→    return (
   357→      <div className="flex min-h-screen flex-col bg-background">
   358→        <TopBar title="창구 반납" />
   359→        <main className="mx-auto flex w-full max-w-lg flex-1 flex-col items-center justify-center gap-6 px-4">
   360→          <CreditCard className="h-16 w-16 text-muted-foreground/50" />
   361→          <h2 className="text-title text-foreground">회원가입이 필요합니다</h2>
   362→          <p className="text-body text-muted-foreground text-center">
   363→            먼저 회원가입을 해주세요
   364→          </p>
   365→          <button
   366→            onClick={() => setView('registration')}
   367→            className="btn-senior bg-primary text-primary-foreground rounded-xl"
   368→          >
   369→            <UserCheck className="h-6 w-6" />
   370→            회원가입하기
   371→          </button>
   372→        </main>
   373→      </div>
   374→    )
   375→  }
   376→
   377→  // Loading state
   378→  if (loading && step === 1) {
   379→    return (
   380→      <div className="flex min-h-screen flex-col bg-background">
   381→        <TopBar title="창구 반납" />
   382→        <main className="mx-auto flex w-full max-w-lg flex-1 flex-col items-center justify-center gap-4">
   383→          <div className="h-16 w-16 rounded-full border-4 border-primary/20 border-t-primary animate-spin" />
   384→          <p className="text-body text-muted-foreground">불러오는 중...</p>
   385→        </main>
   386→      </div>
   387→    )
   388→  }
   389→
   390→  // No active loans
   391→  if (!loading && step === 1 && activeLoans.length === 0) {
   392→    return (
   393→      <div className="flex min-h-screen flex-col bg-background">
   394→        <TopBar title="창구 반납" />
   395→        <main className="mx-auto flex w-full max-w-lg flex-1 flex-col items-center justify-center gap-6 px-4">
   396→          <motion.div
   397→            initial={{ scale: 0 }}
   398→            animate={{ scale: 1 }}
   399→            transition={{ type: 'spring' }}
   400→          >
   401→            <BookOpen className="h-16 w-16 text-muted-foreground/50" />
   402→          </motion.div>
   403→          <h2 className="text-title text-foreground">반납할 도서가 없습니다</h2>
   404→          <p className="text-body text-muted-foreground text-center">
   405→            현재 대출 중인 책이 없습니다.
   406→          </p>
   407→          <button
   408→            onClick={() => setView('home')}
   409→            className="btn-senior bg-primary text-primary-foreground rounded-xl"
   410→          >
   411→            <Home className="h-6 w-6" />
   412→            홈으로
   413→          </button>
   414→        </main>
   415→      </div>
   416→    )
   417→  }
   418→
   419→  return (
   420→    <div className="flex min-h-screen flex-col bg-background">
   421→      <TopBar
   422→        title="창구 반납"
   423→        helpText="창구 반납 화면입니다. 반납할 책을 선택하고 바코드 스캔을 통해 반납하는 연습을 합니다."
   424→      />
   425→
   426→      <main className="mx-auto w-full max-w-lg flex-1 px-4 py-6">
   427→        <StepProgress current={step} total={4} />
   428→
   429→        <AnimatePresence mode="wait">
   430→          {/* Step 1: Select books to return */}
   431→          {step === 1 && (
   432→            <motion.div
   433→              key="step1"
   434→              initial={{ opacity: 0, x: 50 }}
   435→              animate={{ opacity: 1, x: 0 }}
   436→              exit={{ opacity: 0, x: -50 }}
   437→              className="flex flex-col gap-6"
   438→            >
   439→              <h2 className="text-title text-foreground text-center">
   440→                반납할 책 선택
   441→              </h2>
   442→
   443→              {/* Selection counter */}
   444→              {selectedLoans.length > 0 && (
   445→                <div className="rounded-xl bg-primary/5 border-2 border-primary/20 p-4 text-center">
   446→                  <p className="text-heading font-bold text-primary">
   447→                    선택한 책: {selectedLoans.length}권
   448→                  </p>
   449→                </div>
   450→              )}
   451→
   452→              {/* Loans list */}
   453→              <div className="max-h-96 overflow-y-auto custom-scrollbar flex flex-col gap-3">
   454→                {activeLoans.map((loan, index) => {
   455→                  const isSelected = selectedLoans.some((l) => l.id === loan.id)
   456→                  const days = getDaysRemaining(loan.dueDate)
   457→
   458→                  return (
   459→                    <motion.div
   460→                      key={loan.id}
   461→                      initial={{ opacity: 0, y: 10 }}
   462→                      animate={{ opacity: 1, y: 0 }}
   463→                      transition={{ delay: index * 0.03 }}
   464→                      onClick={() => toggleLoan(loan)}
   465→                      className={cn(
   466→                        'card-senior cursor-pointer transition-all',
   467→                        isSelected && 'border-primary bg-primary/5'
   468→                      )}
   469→                    >
   470→                      <div className="flex items-center gap-3">
   471→                        {/* Checkbox */}
   472→                        <div
   473→                          className="flex h-12 w-12 shrink-0 items-center justify-center rounded-lg border-2 transition-colors"
   474→                          style={{
   475→                            borderColor: isSelected ? 'var(--primary)' : 'var(--border)',
   476→                            background: isSelected ? 'var(--primary)' : 'var(--card)',
   477→                          }}
   478→                        >
   479→                          {isSelected && (
   480→                            <CheckCircle className="h-8 w-8 text-primary-foreground" />
   481→                          )}
   482→                        </div>
   483→
   484→                        {/* Book cover */}
   485→                        <div className="flex h-14 w-10 shrink-0 items-center justify-center rounded-lg bg-amber-100">
   486→                          <BookOpen className="h-6 w-6 text-amber-700" />
   487→                        </div>
   488→
   489→                        {/* Info */}
   490→                        <div className="flex-1 min-w-0">
   491→                          <h4 className="text-body font-bold text-foreground truncate">
   492→                            {loan.book?.title || '제목 없음'}
   493→                          </h4>
   494→                          <p className="text-caption text-muted-foreground truncate">
   495→                            {loan.book?.author || ''}
   496→                          </p>
   497→                        </div>
   498→
   499→                        {/* Days badge */}
   500→                        <Badge className={cn('shrink-0', getDaysColor(days))} variant="secondary">
   501→                          {getDaysLabel(days)}
   502→                        </Badge>
   503→                      </div>
   504→
   505→                      {/* Loan details */}
   506→                      <div className="mt-2 flex items-center gap-4 text-caption text-muted-foreground">
   507→                        <span className="flex items-center gap-1">
   508→                          <Clock className="h-3.5 w-3.5" />
   509→                          대출: {formatDate(loan.loanDate)}
   510→                        </span>
   511→                        <span className="flex items-center gap-1">
   512→                          <Clock className="h-3.5 w-3.5" />
   513→                          반납: {formatDate(loan.dueDate)}
   514→                        </span>
   515→                      </div>
   516→                    </motion.div>
   517→                  )
   518→                })}
   519→              </div>
   520→
   521→              {/* Action button */}
   522→              <button
   523→                onClick={runScanSimulation}
   524→                disabled={selectedLoans.length === 0}
   525→                className="btn-senior w-full bg-primary text-primary-foreground rounded-xl"
   526→              >
   527→                <HandCoins className="h-6 w-6" />
   528→                {selectedLoans.length === 0
   529→                  ? '반납할 책을 선택해주세요'
   530→                  : `반납하기 (${selectedLoans.length}권)`
   531→                }
   532→              </button>
   533→            </motion.div>
   534→          )}
   535→
   536→          {/* Step 2: Scan simulation */}
   537→          {step === 2 && (
   538→            <motion.div
   539→              key="step2"
   540→              initial={{ opacity: 0, x: 50 }}
   541→              animate={{ opacity: 1, x: 0 }}
   542→              exit={{ opacity: 0, x: -50 }}
   543→              className="flex flex-col gap-6"
   544→            >
   545→              <h2 className="text-title text-foreground text-center">
   546→                책 스캔 중
   547→              </h2>
   548→
   549→              {/* Progress */}
   550→              <div className="rounded-xl bg-primary/5 border-2 border-primary/20 p-4 text-center">
   551→                <p className="text-heading font-bold text-primary">
   552→                  {scannedIndices.size} / {selectedLoans.length}권 스캔 완료
   553→                </p>
   554→              </div>
   555→
   556→              {/* Scan cards */}
   557→              <div className="max-h-96 overflow-y-auto custom-scrollbar flex flex-col gap-4">
   558→                {selectedLoans.map((loan, index) => (
   559→                  <ScanCard
   560→                    key={loan.id}
   561→                    book={loan}
   562→                    index={index}
   563→                    total={selectedLoans.length}
