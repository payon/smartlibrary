    38→    38→// Format date to Korean
    39→    39→function formatDate(dateStr: string): string {
    40→    40→  const d = new Date(dateStr)
    41→    41→  return `${d.getFullYear()}년 ${d.getMonth() + 1}월 ${d.getDate()}일`
    42→    42→}

    44→    44→// Calculate overdue days
    45→    45→function getOverdueDays(dueDate: string): number {
    46→    46→  const due = new Date(dueDate)
    47→    47→  const now = new Date()
    48→    48→  const diffMs = now.getTime() - due.getTime()
    49→    49→  const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24))
    50→    50→  return diffDays > 0 ? diffDays : 0
    51→    51→}
