   159→## 2. 인증 설계
   160→
   161→### 2.1 인증 아키텍처 개요
   162→
   163→관리자 대시보드는 **JWT 기반 stateless 인증**을 사용한다. Access Token은 단기 수명(15분)으로 API 요청 시 `Authorization` 헤더에 전달하고, Refresh Token은 장기 수명(7일)으로 `httpOnly` 쿠키에 저장하여 자동 갱신한다.
   164→
   165→```
   166→┌──────────┐  POST /api/auth/login  ┌───────────────────┐
   167→│  Login   │ ──────────────────────►│  bcrypt.compare()  │
   168→│  Page    │                        │  ↓                  │
   169→│          │ ◄────────────────────  │  signAccessToken()  │
   170→│          │  Set-Cookie: refresh   │  signRefreshToken() │
   171→└──────────┘                        └───────────────────┘
   172→     │
   173→     ▼  모든 /api/admin/* 요청
   174→┌──────────────────────────────────────────────────────────┐
   175→│  Authorization: Bearer <accessToken>                     │
   176→│  → verifyJwt() → decode role → checkPermission() → 401/403/next │
   177→└──────────────────────────────────────────────────────────┘
   178→     │ accessToken 만료 (15분)
   179→     ▼
   180→┌──────────┐  POST /api/auth/refresh  ┌───────────────────┐
   181→│  Client  │ ────────────────────────►│  verifyRefresh()   │
   182→│  (자동)  │  Cookie: refreshToken    │  ↓                  │
   183→│          │ ◄──────────────────────  │  signAccessToken()  │
   184→│          │  { accessToken }         │  rotateRefresh()    │
   185→└──────────┘                          └───────────────────┘
   186→```

   248→### 2.3 비밀번호 해시 설계
   249→
   250→```typescript
   251→// src/lib/admin/auth.ts
   252→
   253→import bcrypt from 'bcrypt';
   254→
   255→const SALT_ROUNDS = 12;
   256→
   257→/** 비밀번호 해시 생성 (관리자 계정 생성/수정 시) */
   258→export async function hashPassword(plainPassword: string): Promise<string> {
   259→  return bcrypt.hash(plainPassword, SALT_ROUNDS);
   260→}
   261→
   262→/** 비밀번호 검증 (로그인 시) */
   263→export async function verifyPassword(
   264→  plainPassword: string,
   265→  hashedPassword: string
   266→): Promise<boolean> {
   267→  return bcrypt.compare(plainPassword, hashedPassword);
   268→}
   269→
   270→/** 비밀번호 정책 검증 */
   271→export function validatePasswordPolicy(password: string): {
   272→  valid: boolean;
   273→  errors: string[];
   274→} {
   275→  const errors: string[] = [];
   276→
   277→  if (password.length < 8) {
   278→    errors.push('비밀번호는 8자 이상이어야 합니다.');
   279→  }
   280→  if (!/[a-z]/.test(password)) {
   281→    errors.push('소문자를 포함해야 합니다.');
   282→  }
   283→  if (!/[A-Z]/.test(password)) {
   284→    errors.push('대문자를 포함해야 합니다.');
   285→  }
   286→  if (!/[0-9]/.test(password)) {
   287→    errors.push('숫자를 포함해야 합니다.');
   288→  }
   289→  if (!/[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(password)) {
   290→    errors.push('특수문자를 포함해야 합니다.');
   291→  }
   292→
   293→  return { valid: errors.length === 0, errors };
   294→}
   295→```

   297→### 2.4 JWT 발급 및 검증
   298→
   299→```typescript
   300→// src/lib/admin/auth.ts (계속)
   301→
   302→import { SignJWT, jwtVerify } from 'jose';
   303→
   304→const ACCESS_TOKEN_SECRET = new TextEncoder().encode(
   305→  process.env.JWT_ACCESS_SECRET!  // 최소 32바이트 랜덤 문자열
   306→);
   307→const REFRESH_TOKEN_SECRET = new TextEncoder().encode(
   308→  process.env.JWT_REFRESH_SECRET! // 최소 32바이트 랜덤 문자열
   309→);
   310→
   311→const ACCESS_TOKEN_TTL = '15m';   // 15분
   312→const REFRESH_TOKEN_TTL = '7d';   // 7일
   313→
   314→/** Access Token 발급 */
   315→export async function signAccessToken(
   316→  payload: Omit<AccessTokenPayload, 'iat' | 'exp'>
   317→): Promise<string> {
   318→  return new SignJWT({
   319→    sub: payload.sub,
   320→    email: payload.email,
   321→    role: payload.role,
   322→    permissions: payload.permissions,
   323→  })
   324→    .setProtectedHeader({ alg: 'HS256' })
   325→    .setIssuedAt()
   326→    .setExpirationTime(ACCESS_TOKEN_TTL)
   327→    .setIssuer('smart-library-admin')
   328→    .setAudience('admin-api')
   329→    .sign(ACCESS_TOKEN_SECRET);
   330→}
   331→
   332→/** Refresh Token 발급 */
   333→export async function signRefreshToken(
   334→  payload: Omit<RefreshTokenPayload, 'iat' | 'exp'>
   335→): Promise<string> {
   336→  return new SignJWT({
   337→    sub: payload.sub,
   338→    tokenVersion: payload.tokenVersion,
   339→  })
   340→    .setProtectedHeader({ alg: 'HS256' })
   341→    .setIssuedAt()
   342→    .setExpirationTime(REFRESH_TOKEN_TTL)
   343→    .setIssuer('smart-library-admin')
   344→    .setAudience('admin-refresh')
   345→    .sign(REFRESH_TOKEN_SECRET);
   346→}
   347→
   348→/** Access Token 검증 */
   349→export async function verifyAccessToken(
   350→  token: string
   351→): Promise<AccessTokenPayload> {
   352→  const { payload } = await jwtVerify(token, ACCESS_TOKEN_SECRET, {
   353→    issuer: 'smart-library-admin',
   354→    audience: 'admin-api',
   355→  });
   356→  return payload as unknown as AccessTokenPayload;
   357→}
   358→
   359→/** Refresh Token 검증 */
   360→export async function verifyRefreshToken(
   361→  token: string
   362→): Promise<RefreshTokenPayload> {
   363→  const { payload } = await jwtVerify(token, REFRESH_TOKEN_SECRET, {
   364→    issuer: 'smart-library-admin',
   365→    audience: 'admin-refresh',
   366→  });
   367→  return payload as unknown as RefreshTokenPayload;
   368→}
   369→```

   371→### 2.5 Refresh Token 쿠키 설정
   372→
   373→```typescript
   374→// src/lib/admin/auth.ts (계속)
   375→
   376→import { cookies } from 'next/headers';
   377→
   378→/** Refresh Token을 httpOnly 쿠키로 설정 */
   379→export function setRefreshTokenCookie(token: string): void {
   380→  cookies().set('refresh_token', token, {
   381→    httpOnly: true,        // JS 접근 불가 (XSS 방어)
   382→    secure: process.env.NODE_ENV === 'production',  // HTTPS 전용 (개발은 제외)
   383→    sameSite: 'strict',   // CSRF 방어
   384→    path: '/api/auth',    // /api/auth 경로에서만 전송
   385→    maxAge: 7 * 24 * 60 * 60,  // 7일 (초)
   386→  });
   387→}
   388→
   389→/** Refresh Token 쿠키 삭제 (로그아웃) */
   390→export function clearRefreshTokenCookie(): void {
   391→  cookies().delete('refresh_token');
   392→}
   393→```

   395→### 2.6 Token 갱신 흐름
   396→
   397→```typescript
   398→// src/app/api/auth/refresh/route.ts
   399→
   400→import { NextRequest, NextResponse } from 'next/server';
   401→import {
   402→  verifyRefreshToken,
   403→  signAccessToken,
   404→  signRefreshToken,
   405→  setRefreshTokenCookie,
   406→  clearRefreshTokenCookie,
   407→} from '@/lib/admin/auth';
   408→import { prisma } from '@/lib/db';
   409→
   410→export async function POST(request: NextRequest) {
   411→  try {
   412→    // 1. 쿠키에서 Refresh Token 추출
   413→    const refreshToken = request.cookies.get('refresh_token')?.value;
   414→    if (!refreshToken) {
   415→      return NextResponse.json(
   416→        { error: { code: 'REFRESH_TOKEN_INVALID', message: 'Refresh token이 없습니다.' } },
   417→        { status: 401 }
   418→      );
   419→    }
   420→
   421→    // 2. Refresh Token 검증
   422→    const payload = await verifyRefreshToken(refreshToken);
   423→
   424→    // 3. DB에서 관리자 정보 조회 (토큰 버전 확인)
   425→    const admin = await prisma.adminUser.findUnique({
   426→      where: { id: payload.sub },
   427→    });
   428→
   429→    if (!admin || !admin.isActive) {
   430→      clearRefreshTokenCookie();
   431→      return NextResponse.json(
   432→        { error: { code: 'ACCOUNT_INACTIVE', message: '비활성화된 계정입니다.' } },
   433→        { status: 401 }
   434→      );
   435→    }
   436→
   437→    // 4. 토큰 버전 불일치 → 강제 로그아웃
   438→    if (admin.tokenVersion !== payload.tokenVersion) {
   439→      clearRefreshTokenCookie();
   440→      return NextResponse.json(
   441→        { error: { code: 'REFRESH_TOKEN_INVALID', message: '토큰이 무효화되었습니다.' } },
   442→        { status: 401 }
   443→      );
   444→    }
   445→
   446→    // 5. 새 Access Token 발급
   447→    const permissions = getPermissionsForRole(admin.role);
   448→    const accessToken = await signAccessToken({
   449→      sub: admin.id,
   450→      email: admin.email,
   451→      role: admin.role as AdminRole,
   452→      permissions,
   453→    });
   454→
   455→    // 6. Refresh Token Rotation (새 Refresh Token 발급)
   456→    const newRefreshToken = await signRefreshToken({
   457→      sub: admin.id,
   458→      tokenVersion: admin.tokenVersion,
   459→    });
   460→    setRefreshTokenCookie(newRefreshToken);
   461→
   462→    // 7. 응답
   463→    return NextResponse.json({
   464→      accessToken,
   465→      user: {
   466→        id: admin.id,
   467→        email: admin.email,
   468→        name: admin.name,
   469→        role: admin.role,
   470→        permissions,
   471→      },
   472→    });
   473→  } catch {
   474→    clearRefreshTokenCookie();
   475→    return NextResponse.json(
   476→      { error: { code: 'TOKEN_INVALID', message: '토큰 검증에 실패했습니다.' } },
   477→      { status: 401 }
   478→    );
   479→  }
   480→}
   481→```

   483→### 2.7 RBAC 미들웨어
   484→
   485→```typescript
   486→// src/lib/admin/rbac.ts
   487→
   488→import { NextRequest, NextResponse } from 'next/server';
   489→import { verifyAccessToken, AccessTokenPayload, AdminRole } from './auth';
   490→
   491→// ─── 권한 정의 ───────────────────────────────────────────
   492→
   493→/** 시스템 내 모든 권한 식별자 */
   494→export type Permission =
   495→  | 'cms:read'           | 'cms:write'
   496→  | 'books:read'         | 'books:write'
   497→  | 'users:read'         | 'users:write'
   498→  | 'accounts:read'      | 'accounts:write'
   499→  | 'stats:read'         | 'stats:export'
   500→  | 'audit:read'         | 'audit:export'
   501→  | 'notices:read'       | 'notices:write'
   502→  | 'settings:read'      | 'settings:write';
   503→
   504→/** 역할별 권한 매핑 (PRD 4.2 기준) */
   505→const ROLE_PERMISSIONS: Record<AdminRole, Permission[]> = {
   506→  super_admin: [
   507→    'cms:read', 'cms:write',
   508→    'books:read', 'books:write',
   509→    'users:read', 'users:write',
   510→    'accounts:read', 'accounts:write',
   511→    'stats:read', 'stats:export',
   512→    'audit:read', 'audit:export',
   513→    'notices:read', 'notices:write',
   514→    'settings:read', 'settings:write',
   515→  ],
   516→  admin: [
   517→    'cms:read', 'cms:write',
   518→    'books:read', 'books:write',
   519→    'users:read', 'users:write',
   520→    'stats:read', 'stats:export',
   521→    'audit:read',
   522→    'notices:read', 'notices:write',
   523→    'settings:read',
   524→  ],
   525→  operator: [
   526→    'cms:read',
   527→    'books:read',
   528→    'users:read',
   529→    'stats:read',
   530→    'notices:read',
   531→  ],
   532→};
   533→
   534→/** 역할에 해당하는 권한 목록 반환 */
   535→export function getPermissionsForRole(role: AdminRole): Permission[] {
   536→  return ROLE_PERMISSIONS[role] ?? [];
   537→}
   538→
   539→/** 특정 권한 보유 여부 확인 */
   540→export function hasPermission(role: AdminRole, permission: Permission): boolean {
   541→  return ROLE_PERMISSIONS[role]?.includes(permission) ?? false;
   542→}
   543→
   544→// ─── API Route 보호 미들웨어 ─────────────────────────────
   545→
   546→type HandlerFn = (request: NextRequest, context: any) => Promise<NextResponse>;
   547→
   548→/** 관리자 인증 + 권한 검사 래퍼 */
   549→export function withAdminAuth(
   550→  handler: HandlerFn,
   551→  requiredPermission: Permission
   552→): HandlerFn {
   553→  return async (request: NextRequest, context: any) => {
   554→    try {
   555→      // 1. Authorization 헤더에서 Bearer 토큰 추출
   556→      const authHeader = request.headers.get('Authorization');
   557→      if (!authHeader?.startsWith('Bearer ')) {
   558→        return NextResponse.json(
   559→          { error: { code: 'TOKEN_INVALID', message: '인증 토큰이 필요합니다.' } },
   560→          { status: 401 }
   561→        );
   562→      }
   563→
   564→      const token = authHeader.slice(7);
   565→
   566→      // 2. Access Token 검증
   567→      const payload: AccessTokenPayload = await verifyAccessToken(token);
   568→
   569→      // 3. 권한 확인
   570→      if (!payload.permissions.includes(requiredPermission)) {
   571→        return NextResponse.json(
   572→          { error: { code: 'PERMISSION_DENIED', message: '접근 권한이 없습니다.' } },
   573→          { status: 403 }
   574→        );
   575→      }
   576→
   577→      // 4. 핸들러 실행 (페이로드를 context에 주입)
   578→      return handler(request, { ...context, auth: payload });
   579→    } catch (error) {
   580→      // 토큰 만료 또는 무효
   581→      if (error instanceof Error && error.name === 'JWTExpired') {
   582→        return NextResponse.json(
   583→          { error: { code: 'TOKEN_EXPIRED', message: '액세스 토큰이 만료되었습니다.' } },
   584→          { status: 401 }
   585→        );
   586→      }
   587→      return NextResponse.json(
   588→        { error: { code: 'TOKEN_INVALID', message: '토큰 검증에 실패했습니다.' } },
   589→        { status: 401 }
   590→      );
   591→    }
   592→  };
   593→}
   594→```

   655→### 2.9 클라이언트 자동 Token 갱신 인터셉터
   656→
   657→```typescript
   658→// src/lib/admin/api-client.ts
   659→
   660→let refreshPromise: Promise<string> | null = null;
   661→
   662→/** 만료된 Access Token을 자동 갱신하는 fetch 래퍼 */
   663→export async function adminFetch<T>(
   664→  url: string,
   665→  options: RequestInit = {}
   666→): Promise<T> {
   667→  const { accessToken, setAccessToken } = useAuthStore.getState();
   668→
   669→  const headers = new Headers(options.headers);
   670→  headers.set('Authorization', `Bearer ${accessToken}`);
   671→
   672→  let response = await fetch(url, { ...options, headers });
   673→
   674→  // 401 → 자동 갱신 시도
   675→  if (response.status === 401) {
   676→    const body = await response.json();
   677→    if (body.error?.code === 'TOKEN_EXPIRED') {
   678→      // 동시 요청 시 갱신 중복 방지
   679→      const newToken = await (refreshPromise ??= refreshToken());
   680→      refreshPromise = null;
   681→
   682→      headers.set('Authorization', `Bearer ${newToken}`);
   683→      response = await fetch(url, { ...options, headers });
   684→    }
   685→  }
   686→
   687→  if (!response.ok) {
   688→    const error = await response.json();
   689→    throw new AdminApiError(response.status, error.error?.code, error.error?.message);
   690→  }
   691→
   692→  return response.json();
   693→}
   694→
   695→/** Refresh Token으로 새 Access Token 발급 */
   696→async function refreshToken(): Promise<string> {
   697→  const res = await fetch('/api/auth/refresh', { method: 'POST' });
   698→  if (!res.ok) {
   699→    useAuthStore.getState().logout();
   700→    window.location.href = '/admin/login';
   701→    throw new Error('인증이 만료되었습니다. 다시 로그인해주세요.');
   702→  }
   703→  const data = await res.json();
   704→  useAuthStore.getState().setAccessToken(data.accessToken);
   705→  return data.accessToken;
   706→}
   707→
   708→/** 관리자 API 에러 클래스 */
   709→export class AdminApiError extends Error {
   710→  constructor(
   711→    public status: number,
   712→    public code: string,
   713→    message: string
   714→  ) {
   715→    super(message);
   716→    this.name = 'AdminApiError';
   717→  }
   718→}
   719→```

   725→### 3.1 Key-Value 콘텐츠 관리 아키텍처
   726→
   727→CMS 콘텐츠는 **Key-Value 패턴**으로 관리한다. 각 화면의 텍스트·이미지·색상·규칙이 `CmsContent` 테이블의 개별 레코드로 저장되며, `key` 값(예: `"idle.title_text"`)으로 프론트엔드가 조회한다. DB에 값이 없으면 코드에 정의된 기본값(`constants.ts`)을 사용한다.
   728→
   729→```
   730→┌────────────────────┐         ┌──────────────────────────────┐
   731→│  constants.ts      │         │  CmsContent (SQLite)          │
   732→│  (코드 기본값)      │         │  key → value (JSON string)    │
   733→│                    │         │                                │
   734→│  idle.title:       │         │  "idle.title_text" → "도서관"  │
   735→│    "스마트 도서관"  │         │  "menu.borrow_button_color"   │
   736→│                    │         │    → "#2563EB"                 │
   737→└────────────────────┘         └──────────────────────────────┘
   738→         │                                  │
   739→         │  DB 값 없으면 기본값 사용          │  DB 값 있으면 DB 값 우선
   740→         ▼                                  ▼
   741→┌──────────────────────────────────────────────────────────────┐
   742→│  useCmsContent("idle.title_text", "스마트 도서관")             │
   743→│  → "도서관"  (DB 오버라이드)                                   │
   744→│                                                                │
   745→│  useCmsContent("idle.subtitle_text", "도서 대출·반납 키오스크")  │
   746→│  → "도서 대출·반납 키오스크"  (코드 기본값, DB 오버라이드 없음)  │
   747→└──────────────────────────────────────────────────────────────┘
   748→```

   919→### 3.4 마이그레이션 전략: 코드 기본값 → DB 오버라이드
   920→
   921→```
   922→초기 배포:
   923→  CmsContent 테이블 = 비어 있음
   924→  → 모든 key가 constants.ts의 기본값으로 동작
   925→

   927→  → CmsContent에 { key: "idle.title_text", value: '"도서관"' } INSERT
   928→  → 프론트엔드는 DB 값을 우선 사용
   929→
   930→새 버전에서 key가 추가됨 (예: "idle.welcome_animation"):
   931→  → CmsContent에 해당 key 레코드 없음
   932→  → constants.ts의 기본값으로 자동 폴백
   933→  → 별도 마이그레이션 불필요 (코드 기본값이 단일 소스 오브 트루스)
   934→```
   935→
   936→```typescript
   937→// src/app/api/admin/cms/content/route.ts
   938→
   939→import { withAdminAuth } from '@/lib/admin/rbac';
   940→import { CMS_DEFAULTS_MAP } from '@/lib/admin/cms-defaults';
   941→import { prisma } from '@/lib/db';
   942→
   943→/** GET /api/admin/cms/content — 전체 CMS 콘텐츠 조회 */
   944→export const GET = withAdminAuth(async (request, { auth }) => {
   945→  // DB에 저장된 모든 CMS 레코드 조회
   946→  const dbContents = await prisma.cmsContent.findMany();
   947→
   948→  // key → value 맵 구성
   949→  const dbMap = new Map(dbContents.map((c) => [c.key, c.value]));
   950→
   951→  // 기본값과 병합: DB 값 우선, 없으면 기본값
   952→  const merged: Record<string, { value: string; type: string; source: 'db' | 'default' }> = {};
   953→
   954→  for (const [key, defaultValue] of CMS_DEFAULTS_MAP) {
   955→    const dbValue = dbMap.get(key);
   956→    merged[key] = {
   957→      value: dbValue ?? defaultValue,
   958→      type: CMS_FIELD_MAP.get(key)!.type,
   959→      source: dbValue ? 'db' : 'default',
   960→    };
   961→  }
   962→
   963→  return NextResponse.json({ content: merged });
   964→}, 'cms:read');
   965→```

   971→
   972→import { useCmsStore } from '@/stores/admin/useCmsStore';
   973→import { CMS_DEFAULTS_MAP } from '@/lib/admin/cms-defaults';

   976→ * 단일 CMS 콘텐츠 값 조회 훅

  1054→## 4. 실시간 동기화 설계
  1055→
  1056→### 4.1 키오스크 폴링 메커니즘 개요
  1057→
  1058→키오스크 프론트엔드는 **3초 간격 폴링**으로 CMS 콘텐츠 변경을 감지한다. 서버는 `ETag` 헤더로 콘텐츠 버전을 관리하고, 클라이언트는 `If-None-Match` 헤더로 조건부 요청을 보내어 변경이 없을 경우 `304 Not Modified`를 수신하여 트래픽을 최소화한다.
  1059→
  1060→```
  1061→┌──────────────┐  GET /api/cms/content        ┌──────────────┐
  1062→│  Kiosk       │  If-None-Match: "etag_v42"   │  Server      │
  1063→│  Frontend    │ ────────────────────────────► │  (Next.js)   │
  1064→│              │                               │              │
  1065→│              │ ◄── 200 + ETag: "etag_v43"   │  (변경 있음)  │
  1066→│              │     { content: {...} }        │              │
  1067→│              │                               │              │
  1068→│  (3초 후)    │  GET /api/cms/content        │              │
  1069→│              │  If-None-Match: "etag_v43"   │              │
  1070→│              │ ────────────────────────────► │              │
  1071→│              │                               │              │
  1072→│              │ ◄── 304 Not Modified          │  (변경 없음)  │
  1073→└──────────────┘                               └──────────────┘
  1074→```
  1075→
  1076→### 4.2 서버: ETag 기반 공개 CMS API
  1077→
  1078→```typescript
  1079→// src/app/api/cms/content/route.ts (키오스크용 공개 API — 인증 불필요)
  1080→
  1081→import { NextRequest, NextResponse } from 'next/server';
  1082→import { prisma } from '@/lib/db';
  1083→import { CMS_DEFAULTS_MAP, CMS_FIELD_MAP } from '@/lib/admin/cms-defaults';
  1084→import { createHash } from 'crypto';
  1085→
  1086→export async function GET(request: NextRequest) {
  1087→  // 1. DB에서 모든 CMS 레코드 조회
  1088→  const dbContents = await prisma.cmsContent.findMany({
  1089→    orderBy: { key: 'asc' },
  1090→  });
  1091→
  1092→  // 2. 기본값과 병합
  1093→  const merged: Record<string, string> = {};
  1094→  for (const [key, defaultValue] of CMS_DEFAULTS_MAP) {
  1095→    const dbRecord = dbContents.find((c) => c.key === key);
  1096→    merged[key] = dbRecord?.value ?? defaultValue;
  1097→  }
  1098→
  1099→  // 3. ETag 생성 (내용 해시)
  1100→  const contentHash = createHash('md5')
  1101→    .update(JSON.stringify(merged))
  1102→    .digest('hex');
  1103→  const etag = `"${contentHash}"`;
  1104→
  1105→  // 4. 조건부 요청 검사 (If-None-Match)
  1106→  const ifNoneMatch = request.headers.get('If-None-Match');
  1107→  if (ifNoneMatch === etag) {
  1108→    return new NextResponse(null, {
  1109→      status: 304,
  1110→      headers: { ETag: etag },
  1111→    });
  1112→  }
  1113→
  1114→  // 5. 변경 있음 → 200 + ETag + 콘텐츠
  1115→  return NextResponse.json(
  1116→    { content: merged, updatedAt: new Date().toISOString() },
  1117→    {
  1118→      headers: {
  1119→        ETag: etag,
  1120→        'Cache-Control': 'no-cache',  // 항상 재검증
  1121→      },
  1122→    }
  1123→  );
  1124→}
  1125→```
  1126→
  1127→### 4.3 클라이언트: useCmsPolling 훅
  1128→
  1129→```typescript
  1130→// src/hooks/admin/useCmsPolling.ts
  1131→
  1132→import { useEffect, useRef, useCallback } from 'react';
  1133→import { useCmsStore } from '@/stores/admin/useCmsStore';
  1134→
  1135→const POLLING_INTERVAL_MS = 3000;  // 3초
  1136→const MAX_RETRIES = 5;
  1137→const RETRY_BASE_DELAY_MS = 1000;
  1138→
  1139→/**
  1140→ * 키오스크용 CMS 콘텐츠 폴링 훅
  1141→ *
  1142→ * 3초 간격으로 GET /api/cms/content를 호출하고,
  1143→ * ETag 기반 조건부 요청으로 불필요한 데이터 전송을 방지한다.
  1144→ * 연결 실패 시 지수 백오프로 재시도한다.
  1145→ */
  1146→export function useCmsPolling(enabled: boolean = true) {
  1147→  const etagRef = useRef<string | null>(null);
  1148→  const retryCountRef = useRef(0);
  1149→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1150→  const setContent = useCmsStore((state) => state.setContent);
  1151→  const setLastUpdated = useCmsStore((state) => state.setLastUpdated);
  1152→  const setConnectionStatus = useCmsStore((state) => state.setConnectionStatus);
  1153→
  1154→  const poll = useCallback(async () => {
  1155→    try {
  1156→      const headers: Record<string, string> = {};
  1157→      if (etagRef.current) {
  1158→        headers['If-None-Match'] = etagRef.current;
  1159→      }
  1160→
  1161→      const response = await fetch('/api/cms/content', { headers });
  1162→
  1163→      if (response.status === 304) {
  1164→        // 변경 없음 — 재시도 카운터 리셋
  1165→        retryCountRef.current = 0;
  1166→        setConnectionStatus('connected');
  1167→        return;
  1168→      }
  1169→
  1170→      if (!response.ok) throw new Error(`HTTP ${response.status}`);
  1171→
  1172→      // ETag 저장
  1173→      const newEtag = response.headers.get('ETag');
  1174→      if (newEtag) etagRef.current = newEtag;
  1175→
  1176→      // 콘텐츠 업데이트
  1177→      const data = await response.json();
  1178→      setContent(new Map(Object.entries(data.content)));
  1179→      setLastUpdated(new Date(data.updatedAt));
  1180→      setConnectionStatus('connected');
  1181→
  1182→      // 재시도 카운터 리셋
  1183→      retryCountRef.current = 0;
  1184→    } catch (error) {
  1185→      retryCountRef.current++;
  1186→      setConnectionStatus('disconnected');
  1187→
  1188→      if (retryCountRef.current >= MAX_RETRIES) {
  1189→        setConnectionStatus('failed');
  1190→        console.error('[CMS Polling] 최대 재시도 횟수 초과:', error);
  1191→        return;
  1192→      }
  1193→
  1194→      // 지수 백오프 대기 후 재시도
  1195→      const delay = RETRY_BASE_DELAY_MS * Math.pow(2, retryCountRef.current - 1);
  1196→      console.warn(`[CMS Polling] 재시도 ${retryCountRef.current}/${MAX_RETRIES} (${delay}ms 후)`);
  1197→      await new Promise((r) => setTimeout(r, delay));
  1198→    }
  1199→  }, [setContent, setLastUpdated, setConnectionStatus]);
  1200→
  1201→  useEffect(() => {
  1202→    if (!enabled) return;
  1203→
  1204→    // 최초 1회 즉시 실행
  1205→    poll();
  1206→
  1207→    // 이후 3초 간격 폴링
  1208→    intervalRef.current = setInterval(poll, POLLING_INTERVAL_MS);
  1209→
  1210→    return () => {
  1211→      if (intervalRef.current) {
  1212→        clearInterval(intervalRef.current);
  1213→      }
  1214→    };
  1215→  }, [enabled, poll]);
  1216→}
  1217→```

  1288→      type: fieldDef.type,
  1289→      value,

  1293→
  1294→  // 감사 로그 기록
  1295→  await recordAuditLog({
  1296→    action: 'UPDATE',
  1297→    entity: 'CmsContent',
  1298→    entityId: result.id,
  1299→    oldValue: fieldDef.defaultValue,
  1300→    newValue: value,
  1301→    performedBy: auth.sub,

  1378→
  1379→/** 매직 넘버 시그니처 */
  1380→export const IMAGE_MAGIC_NUMBERS: Record<string, number[]> = {
  1381→  'image/jpeg':  [0xFF, 0xD8, 0xFF],    // JPEG
  1382→  'image/png':   [0x89, 0x50, 0x4E, 0x47], // PNG
  1383→  'image/webp':  [0x52, 0x49, 0x46, 0x46], // RIFF (WebP container)
  1384→  'image/svg+xml': [],                    // SVG는 텍스트 — 별도 검증
  1385→};
  1386→```

  1388→### 5.3 파일명 새니타이징 및 저장 경로
  1389→
  1390→```typescript
  1391→// src/lib/admin/upload.ts
  1392→
  1393→import path from 'path';
  1394→import fs from 'fs/promises';
  1395→import crypto from 'crypto';
  1396→
  1397→const UPLOAD_DIR = path.join(process.cwd(), 'public', 'uploads', 'cms');
  1398→
  1399→/**
  1400→ * 파일명 새니타이징
  1401→ * - 한글/특수문자 제거
  1402→ * - 공백 → 하이픈
  1403→ * - 타임스탬프 프리픽스로 유일성 보장
  1404→ */
  1405→export function sanitizeFilename(originalName: string): string {
  1406→  const sanitized = originalName
  1407→    .normalize('NFD')
  1408→    .replace(/[\u0300-\u036f]/g, '')  // 분해된 악센트 제거
  1409→    .replace(/[^a-zA-Z0-9._-]/g, '_') // 안전한 문자만 유지
  1410→    .replace(/_+/g, '_')              // 연속 밑줄 → 단일
  1411→    .toLowerCase();
  1412→
  1413→  const timestamp = Date.now();
  1414→  return `${timestamp}_${sanitized}`;
  1415→}
  1416→
  1417→/**
  1418→ * 이미지 파일 저장
  1419→ * @returns 저장된 파일의 공개 URL 경로
  1420→ */
  1421→export async function saveImageFile(
  1422→  buffer: Buffer,
  1423→  originalName: string
  1424→): Promise<string> {
  1425→  // 업로드 디렉터리 보장
  1426→  await fs.mkdir(UPLOAD_DIR, { recursive: true });
  1427→
  1428→  const filename = sanitizeFilename(originalName);
  1429→  const filePath = path.join(UPLOAD_DIR, filename);
  1430→
  1431→  // 파일 저장
  1432→  await fs.writeFile(filePath, buffer);
  1433→
  1434→  // 공개 URL 반환 (Next.js가 /public 아래를 정적 서빙)
  1435→  return `/uploads/cms/${filename}`;
  1436→}
  1437→```

  1439→### 5.4 매직 넘버 검증 (MIME 스푸핑 방어)
  1440→
  1441→```typescript
  1442→// src/lib/admin/upload.ts (계속)
  1443→
  1444→import { IMAGE_MAGIC_NUMBERS, ALLOWED_IMAGE_MIMES, MAX_IMAGE_SIZE } from '@/types/admin';
  1445→
  1446→/**
  1447→ * 파일 매직 넘버 검증
  1448→ * MIME 헤더만 신뢰하지 않고, 파일의 실제 바이트 시그니처를 확인한다.
  1449→ */
  1450→export function validateMagicNumber(buffer: Buffer, claimedMime: string): boolean {
  1451→  // SVG는 텍스트 포맷 — <svg 태그로 시작하는지 확인
  1452→  if (claimedMime === 'image/svg+xml') {
  1453→    const head = buffer.toString('utf8', 0, 256).trim();
  1454→    return head.startsWith('<?xml') || head.startsWith('<svg');
  1455→  }
  1456→
  1457→  const expected = IMAGE_MAGIC_NUMBERS[claimedMime];
  1458→  if (!expected || expected.length === 0) return false;
  1459→
  1460→  // 파일 시작 바이트와 매직 넘버 비교
  1461→  for (let i = 0; i < expected.length; i++) {
  1462→    if (buffer[i] !== expected[i]) return false;
  1463→  }
  1464→
  1465→  // WebP는 RIFF + WEBP 시그니처 확인
  1466→  if (claimedMime === 'image/webp') {
  1467→    const webpSig = buffer.toString('ascii', 8, 12);
  1468→    if (webpSig !== 'WEBP') return false;
  1469→  }
  1470→
  1471→  return true;
  1472→}
  1473→
  1474→/** 이미지 업로드 전체 검증 */
  1475→export function validateImageUpload(
  1476→  buffer: Buffer,
  1477→  claimedMime: string,
  1478→  fileSize: number
  1479→): { valid: boolean; error?: string } {
  1480→  // 1. MIME 타입 허용 여부
  1481→  if (!ALLOWED_IMAGE_MIMES.includes(claimedMime as any)) {
  1482→    return { valid: false, error: `허용되지 않는 파일 형식: ${claimedMime}` };
  1483→  }
  1484→
  1485→  // 2. 파일 크기 제한
  1486→  if (fileSize > MAX_IMAGE_SIZE) {
  1487→    return { valid: false, error: `파일 크기 초과: ${fileSize} > ${MAX_IMAGE_SIZE} bytes` };
  1488→  }
  1489→
  1490→  // 3. 매직 넘버 검증
  1491→  if (!validateMagicNumber(buffer, claimedMime)) {
  1492→    return { valid: false, error: '파일 시그니처가 MIME 타입과 일치하지 않습니다.' };
  1493→  }
  1494→
  1495→  return { valid: true };
  1496→}
  1497→```

  1499→### 5.5 썸네일 생성 (sharp)
  1500→
  1501→```typescript
  1502→// src/lib/admin/upload.ts (계속)
  1503→
  1504→import sharp from 'sharp';
  1505→
  1506→const THUMBNAIL_WIDTH = 200;
  1507→const THUMBNAIL_HEIGHT = 200;
  1508→
  1509→/**
  1510→ * 썸네일 이미지 생성
  1511→ * 원본 비율 유지, 최대 200×200으로 축소, WebP 포맷 변환
  1512→ */
  1513→export async function generateThumbnail(
  1514→  buffer: Buffer,
  1515→  originalName: string
  1516→): Promise<string> {
  1517→  // SVG는 sharp 처리 불가 — 원본 URL을 썸네일로 사용
  1518→  if (originalName.endsWith('.svg')) {
  1519→    return ''; // 호출부에서 원본 URL 사용
  1520→  }
  1521→
  1522→  const thumbnailBuffer = await sharp(buffer)
  1523→    .resize(THUMBNAIL_WIDTH, THUMBNAIL_HEIGHT, {
  1524→      fit: 'inside',    // 비율 유지
  1525→      withoutEnlargement: true,  // 작은 이미지 확대 금지
  1526→    })
  1527→    .webp({ quality: 80 })
  1528→    .toBuffer();
  1529→
  1530→  const thumbFilename = `thumb_${sanitizeFilename(originalName).replace(/\.[^.]+$/, '.webp')}`;
  1531→  const thumbPath = path.join(UPLOAD_DIR, thumbFilename);
  1532→  await fs.writeFile(thumbPath, thumbnailBuffer);
  1533→
  1534→  return `/uploads/cms/${thumbFilename}`;
  1535→}
  1536→```

  1538→### 5.6 이미지 업로드 API Route
  1539→
  1540→```typescript
  1541→// src/app/api/admin/cms/upload/route.ts
  1542→
  1543→import { NextRequest, NextResponse } from 'next/server';
  1544→import { withAdminAuth } from '@/lib/admin/rbac';
  1545→import { validateImageUpload, saveImageFile, generateThumbnail } from '@/lib/admin/upload';
  1546→import formidable from 'formidable';
  1547→
  1548→export const POST = withAdminAuth(async (request, { auth }) => {
  1549→  // 1. 멀티파트 폼 데이터 파싱
  1550→  const formData = await request.formData();
  1551→  const file = formData.get('file') as File | null;
  1552→  const key = formData.get('key') as string;
  1553→  const altText = formData.get('altText') as string | null;
  1554→
  1555→  if (!file || !key) {
  1556→    return NextResponse.json(
  1557→      { error: { code: 'BAD_REQUEST', message: 'file과 key는 필수입니다.' } },
  1558→      { status: 400 }
  1559→    );
  1560→  }
  1561→
  1562→  // 2. 파일 버퍼 읽기
  1563→  const buffer = Buffer.from(await file.arrayBuffer());
  1564→
  1565→  // 3. 검증
  1566→  const validation = validateImageUpload(buffer, file.type, file.size);
  1567→  if (!validation.valid) {
  1568→    return NextResponse.json(
  1569→      { error: { code: 'INVALID_FILE', message: validation.error } },
  1570→      { status: 400 }
  1571→    );
  1572→  }
  1573→
  1574→  // 4. 파일 저장
  1575→  const url = await saveImageFile(buffer, file.name);
  1576→
  1577→  // 5. 썸네일 생성
  1578→  const thumbnailUrl = await generateThumbnail(buffer, file.name);
  1579→
  1580→  // 6. CMS 콘텐츠 업데이트
  1581→  await prisma.cmsContent.upsert({
  1582→    where: { key },
  1583→    update: { value: JSON.stringify({ url, thumbnailUrl, altText }), updatedBy: auth.sub },
  1584→    create: { key, type: 'image', value: JSON.stringify({ url, thumbnailUrl, altText }), updatedBy: auth.sub },
  1585→  });
  1586→
  1587→  // 7. 감사 로그
  1588→  await recordAuditLog({
  1589→    action: 'UPDATE',
  1590→    entity: 'CmsContent',
  1591→    entityId: key,
  1592→    newValue: url,
  1593→    performedBy: auth.sub,
  1594→  });
  1595→
  1596→  return NextResponse.json({
  1597→    url,
  1598→    thumbnailUrl: thumbnailUrl || url,  // SVG는 원본 URL
  1599→    altText,
  1600→    size: file.size,
  1601→  });
  1602→}, 'cms:write');
  1603→```

  1618→export async function updateImageAltText(
  1619→  key: string,
  1620→  altText: string,
  1621→  adminId: string
  1622→): Promise<void> {

  1628→
  1629→  await prisma.cmsContent.update({
  1630→    where: { key },
  1631→    data: { value: JSON.stringify(current), updatedBy: adminId },
  1632→  });
  1633→}
