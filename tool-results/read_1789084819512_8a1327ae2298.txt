   111→   111→### 1.3 공통 의사코드 (TypeScript)
   112→   112→
   113→   113→```typescript
   114→   114→// lib/admin-handler.ts
   115→   115→type AdminHandlerContext = {
   116→   116→  admin: AdminUser;          // JWT에서 추출한 관리자 정보
   117→   117→  permissions: string[];     // 역할에서 전개한 권한 목록
   118→   118→};
   119→   119→
   120→   120→type AdminHandler<TInput, TOutput> = (
   121→   121→  ctx: AdminHandlerContext,
   122→   122→  input: TInput
   123→   123→) => Promise<TOutput>;
   124→   124→
   125→   125→async function adminApiHandler<TInput, TOutput>(
   126→   126→  request: NextRequest,
   127→   127→  config: {
   128→   128→    requiredPermission: string;
   129→   129→    validateSchema: ZodSchema<TInput>;
   130→   130→    handler: AdminHandler<TInput, TOutput>;
   131→   131→    auditAction?: string;
   132→   132→    auditEntity?: string;
   133→   133→  }
   134→   134→): Promise<NextResponse> {
   135→   135→  // Step 1: JWT 검증
   136→   136→  const token = request.cookies.get('admin_access_token')?.value;
   137→   137→  if (!token) return NextResponse.json(
   138→   138→    { error: { code: 'UNAUTHORIZED', message: '토큰이 없습니다.' } },
   139→   139→    { status: 401 }
   140→   140→  );
   141→   141→
   142→   142→  const payload = verifyJwt(token);
   143→   143→  if (!payload) return NextResponse.json(
   144→   144→    { error: { code: 'UNAUTHORIZED', message: '토큰이 만료되거나 무효합니다.' } },
   145→   145→    { status: 401 }
   146→   146→  );
   147→   147→
   148→   148→  // Step 2: 관리자 조회 + 역할 추출
   149→   149→  const admin = await prisma.adminUser.findUnique({
   150→   150→    where: { id: payload.sub },
   151→   151→    include: { role: { include: { permissions: true } } }
   152→   152→  });
   153→   153→  if (!admin || !admin.isActive) return NextResponse.json(
   154→   154→    { error: { code: 'UNAUTHORIZED', message: '비활성화된 계정입니다.' } },
   155→   155→    { status: 401 }
   156→   156→  );
   157→   157→
   158→   158→  // Step 3: RBAC 권한 검사
   159→   159→  const permissions = admin.role.permissions.map(p => p.code);
   160→   160→  if (!permissions.includes(config.requiredPermission)) {
   161→   161→    return NextResponse.json(
   162→   162→      { error: { code: 'FORBIDDEN', message: '권한이 부족합니다.' } },
   163→   163→      { status: 403 }
   164→   164→    );
   165→   165→  }
   166→   166→
   167→   167→  // Step 4: 입력 검증 (Zod)
   168→   168→  const body = await request.json();
   169→   169→  const parsed = config.validateSchema.safeParse(body);
   170→   170→  if (!parsed.success) return NextResponse.json(
   171→   171→    { error: { code: 'VALIDATION_ERROR', message: '입력값이 유효하지 않습니다.', details: parsed.error.flatten() } },
   172→   172→    { status: 400 }
   173→   173→  );
   174→   174→
   175→   175→  // Step 5: 비즈니스 로직 실행
   176→   176→  const ctx: AdminHandlerContext = { admin, permissions };
   177→   177→  const result = await config.handler(ctx, parsed.data);
   178→   178→
   179→   179→  // Step 6: 감사 로그 (쓰기 작업만)
   180→   180→  if (config.auditAction) {
   181→   181→    await writeAuditLog({
   182→   182→      action: config.auditAction,
   183→   183→      entity: config.auditEntity ?? 'Unknown',
   184→   184→      entityId: result.id,
   185→   185→      oldValue: result.oldValue ?? null,
   186→   186→      newValue: result.newValue ?? null,
   187→   187→      performedBy: admin.id,
   188→   188→    });
   189→   189→  }
   190→   190→
   191→   191→  // Step 7: 응답
   192→   192→  return NextResponse.json({ data: result.data }, { status: result.status ?? 200 });
   193→   193→}
   194→   194→```

   249→   249→#### 2.1.1 로그인 의사코드
   250→   250→
   251→   251→```typescript
   252→   252→// app/api/admin/auth/login/route.ts
   253→   253→async function handleLogin(request: NextRequest): Promise<NextResponse> {
   254→   254→  const { email, password } = LoginSchema.parse(await request.json());
   255→   255→
   256→   256→  // 1. 이메일로 관리자 조회
   257→   257→  const admin = await prisma.adminUser.findUnique({
   258→   258→    where: { email },
   259→   259→    include: { role: true }
   260→   260→  });
   261→   261→
   262→   262→  // 2. 계정 존재 + 활성 + 잠금 상태 확인
   263→   263→  const LOGIN_FAIL_MESSAGE = '이메일 또는 비밀번호가 일치하지 않습니다.';
   264→   264→  if (!admin || !admin.isActive) {
   265→   265→    return errorResponse(401, 'AUTH_FAILED', LOGIN_FAIL_MESSAGE);
   266→   266→  }
   267→   267→
   268→   268→  // 잠금 확인 (5회 연속 실패 → 15분 잠금)
   269→   269→  if (admin.lockedUntil && admin.lockedUntil > new Date()) {
   270→   270→    return errorResponse(401, 'ACCOUNT_LOCKED', '계정이 잠겼습니다. 잠시 후 다시 시도하세요.');
   271→   271→  }
   272→   272→
   273→   273→  // 3. bcrypt 비밀번호 비교
   274→   274→  const isMatch = await bcrypt.compare(password, admin.passwordHash);
   275→   275→  if (!isMatch) {
   276→   276→    // 실패 횟수 증가
   277→   277→    const newFailCount = admin.loginFailCount + 1;
   278→   278→    const lockUpdate = newFailCount >= 5
   279→   279→      ? { loginFailCount: newFailCount, lockedUntil: addMinutes(new Date(), 15) }
   280→   280→      : { loginFailCount: newFailCount };
   281→   281→
   282→   282→    await prisma.adminUser.update({
   283→   283→      where: { id: admin.id },
   284→   284→      data: lockUpdate
   285→   285→    });
   286→   286→
   287→   287→    await writeAuditLog({
   288→   288→      action: 'LOGIN_FAILED',
   289→   289→      entity: 'AdminUser',
   290→   290→      entityId: admin.id,
   291→   291→      performedBy: admin.id,
   292→   292→    });
   293→   293→
   294→   294→    return errorResponse(401, 'AUTH_FAILED', LOGIN_FAIL_MESSAGE);
   295→   295→  }
   296→   296→
   297→   297→  // 4. JWT 생성
   298→   298→  const accessToken = signJwt(
   299→   299→    { sub: admin.id, role: admin.role.code, permissions: admin.role.permissions },
   300→   300→    { expiresIn: '15m' }
   301→   301→  );
   302→   302→  const refreshToken = signJwt(
   303→   303→    { sub: admin.id, type: 'refresh' },
   304→   304→    { expiresIn: '7d' }
   305→   305→  );
   306→   306→
   307→   307→  // 5. 실패 횟수 초기화 + 마지막 로그인 시간 갱신
   308→   308→  await prisma.adminUser.update({
   309→   309→    where: { id: admin.id },
   310→   310→    data: {
   311→   311→      loginFailCount: 0,
   312→   312→      lockedUntil: null,
   313→   313→      lastLoginAt: new Date(),
   314→   314→      refreshTokenHash: hashToken(refreshToken), // refresh token 저장 (순환 검증용)
   315→   315→    }
   316→   316→  });
   317→   317→
   318→   318→  // 6. 감사 로그
   319→   319→  await writeAuditLog({
   320→   320→    action: 'LOGIN',
   321→   321→    entity: 'AdminUser',
   322→   322→    entityId: admin.id,
   323→   323→    newValue: { lastLoginAt: new Date().toISOString() },
   324→   324→    performedBy: admin.id,
   325→   325→  });
   326→   326→
   327→   327→  // 7. httpOnly 쿠키 설정 + 응답
   328→   328→  const response = NextResponse.json({
   329→   329→    data: {
   330→   330→      user: { id: admin.id, name: admin.name, email: admin.email },
   331→   331→      role: admin.role.code,
   332→   332→      permissions: admin.role.permissions.map(p => p.code),
   333→   333→    }
   334→   334→  });
   335→   335→
   336→   336→  response.cookies.set('admin_access_token', accessToken, {
   337→   337→    httpOnly: true,
   338→   338→    secure: process.env.NODE_ENV === 'production',
   339→   339→    sameSite: 'lax',
   340→   340→    path: '/',
   341→   341→    maxAge: 15 * 60, // 15분
   342→   342→  });
   343→   343→
   344→   344→  response.cookies.set('admin_refresh_token', refreshToken, {
   345→   345→    httpOnly: true,
   346→   346→    secure: process.env.NODE_ENV === 'production',
   347→   347→    sameSite: 'lax',
   348→   348→    path: '/api/admin/auth/refresh', // refresh 엔드포인트에서만 전송
   349→   349→    maxAge: 7 * 24 * 60 * 60, // 7일
   350→   350→  });
   351→   351→
   352→   352→  return response;
   353→   353→}
   354→   354→```

   365→   365→### 2.2 세션 검증 미들웨어
   366→   366→
   367→   367→모든 `/api/admin/*` 요청에 대해 JWT를 검증하고 관리자 정보를 주입하는 미들웨어.
   368→   368→
   369→   369→```typescript
   370→   370→// middleware.ts (Next.js middleware)
   371→   371→export async function middleware(request: NextRequest) {
   372→   372→  // 관리자 API 경로만 처리
   373→   373→  if (!request.nextUrl.pathname.startsWith('/api/admin/')) {
   374→   374→    return NextResponse.next();
   375→   375→  }
   376→   376→
   377→   377→  // 로그인·리프레시 엔드포인트는 제외
   378→   378→  if (
   379→   379→    request.nextUrl.pathname === '/api/admin/auth/login' ||
   380→   380→    request.nextUrl.pathname === '/api/admin/auth/refresh'
   381→   381→  ) {
   382→   382→    return NextResponse.next();
   383→   383→  }
   384→   384→
   385→   385→  // Step 1: access token 추출
   386→   386→  const accessToken = request.cookies.get('admin_access_token')?.value;
   387→   387→  if (!accessToken) {
   388→   388→    return NextResponse.json(
   389→   389→      { error: { code: 'UNAUTHORIZED', message: '인증이 필요합니다.' } },
   390→   390→      { status: 401 }
   391→   391→    );
   392→   392→  }
   393→   393→
   394→   394→  // Step 2: JWT 검증
   395→   395→  const payload = verifyJwt(accessToken);
   396→   396→  if (!payload) {
   397→   397→    return NextResponse.json(
   398→   398→      { error: { code: 'TOKEN_EXPIRED', message: '액세스 토큰이 만료되었습니다.' } },
   399→   399→      { status: 401 }
   400→   400→    );
   401→   401→  }
   402→   402→
   403→   403→  // Step 3: 관리자 활성 상태 확인
   404→   404→  const admin = await prisma.adminUser.findUnique({
   405→   405→    where: { id: payload.sub },
   406→   406→    select: { id: true, isActive: true, role: { include: { permissions: true } } }
   407→   407→  });
   408→   408→
   409→   409→  if (!admin || !admin.isActive) {
   410→   410→    return NextResponse.json(
   411→   411→      { error: { code: 'UNAUTHORIZED', message: '비활성화된 계정입니다.' } },
   412→   412→      { status: 401 }
   413→   413→    );
   414→   414→  }
   415→   415→
   416→   416→  // Step 4: 요청 헤더에 관리자 정보 주입 (다운스트림 핸들러에서 사용)
   417→   417→  const requestHeaders = new Headers(request.headers);
   418→   418→  requestHeaders.set('x-admin-id', admin.id);
   419→   419→  requestHeaders.set('x-admin-role', admin.role.code);
   420→   420→  requestHeaders.set('x-admin-permissions', JSON.stringify(admin.role.permissions.map(p => p.code)));
   421→   421→
   422→   422→  return NextResponse.next({

   424→   424→  });
   425→   425→}

   429→   429→};

   472→   472→#### 2.3.1 토큰 갱신 의사코드
   473→   473→
   474→   474→```typescript
   475→   475→// app/api/admin/auth/refresh/route.ts
   476→   476→async function handleRefresh(request: NextRequest): Promise<NextResponse> {
   477→   477→  const refreshToken = request.cookies.get('admin_refresh_token')?.value;
   478→   478→  if (!refreshToken) {
   479→   479→    return errorResponse(401, 'NO_REFRESH_TOKEN', '리프레시 토큰이 없습니다.');
   480→   480→  }
   481→   481→
   482→   482→  // 1. JWT 검증
   483→   483→  const payload = verifyJwt(refreshToken, { allowExpired: false });
   484→   484→  if (!payload || payload.type !== 'refresh') {
   485→   485→    return errorResponse(401, 'INVALID_REFRESH_TOKEN', '유효하지 않은 리프레시 토큰입니다.');
   486→   486→  }
   487→   487→
   488→   488→  // 2. DB에 저장된 해시와 일치 확인 (토큰 재사용 공격 방지)
   489→   489→  const admin = await prisma.adminUser.findUnique({
   490→   490→    where: { id: payload.sub },
   491→   491→    select: { id: true, isActive: true, refreshTokenHash: true, role: true }
   492→   492→  });
   493→   493→
   494→   494→  if (!admin || !admin.isActive || admin.refreshTokenHash !== hashToken(refreshToken)) {
   495→   495→    // 의심스러운 재사용 → 모든 토큰 무효화
   496→   496→    await prisma.adminUser.update({
   497→   497→      where: { id: payload.sub },
   498→   498→      data: { refreshTokenHash: null }
   499→   499→    });
   500→   500→    return errorResponse(401, 'TOKEN_REUSED', '토큰이 무효화되었습니다. 다시 로그인하세요.');
   501→   501→  }
   502→   502→
   503→   503→  // 3. 새 access token 생성
   504→   504→  const newAccessToken = signJwt(
   505→   505→    { sub: admin.id, role: admin.role.code },
   506→   506→    { expiresIn: '15m' }
   507→   507→  );
   508→   508→
   509→   509→  // 4. 쿠키 설정
   510→   510→  const response = NextResponse.json({ success: true });
   511→   511→  response.cookies.set('admin_access_token', newAccessToken, {
   512→   512→    httpOnly: true,
   513→   513→    secure: process.env.NODE_ENV === 'production',
   514→   514→    sameSite: 'lax',
   515→   515→    path: '/',
   516→   516→    maxAge: 15 * 60,
   517→   517→  });
   518→   518→
   519→   519→  return response;
   520→   520→}
   521→   521→```

   523→   523→#### 2.3.2 클라이언트 측 자동 갱신 인터셉터
   524→   524→
   525→   525→```typescript
   526→   526→// lib/admin-fetch.ts
   527→   527→let isRefreshing = false;
   528→   528→let pendingRequests: Array<() => void> = [];
   529→   529→
   530→   530→async function adminFetch<T>(url: string, options?: RequestInit): Promise<T> {
   531→   531→  const response = await fetch(url, { ...options, credentials: 'include' });
   532→   532→
   533→   533→  // 401 + TOKEN_EXPIRED → 자동 갱신 시도
   534→   534→  if (response.status === 401) {
   535→   535→    const body = await response.json();
   536→   536→    if (body.error?.code === 'TOKEN_EXPIRED') {
   537→   537→      // 이미 갱신 중이면 대기
   538→   538→      if (isRefreshing) {
   539→   539→        await new Promise<void>(resolve => pendingRequests.push(resolve));
   540→   540→        return adminFetch<T>(url, options); // 원래 요청 재시도
   541→   541→      }
   542→   542→
   543→   543→      isRefreshing = true;
   544→   544→      try {
   545→   545→        const refreshRes = await fetch('/api/admin/auth/refresh', {
   546→   546→          method: 'POST',
   547→   547→          credentials: 'include',
   548→   548→        });
   549→   549→
   550→   550→        if (!refreshRes.ok) {
   551→   551→          // 갱신 실패 → 강제 로그아웃
   552→   552→          window.location.href = '/admin/login';
   553→   553→          throw new Error('세션이 만료되었습니다.');
   554→   554→        }
   555→   555→
   556→   556→        // 대기 중인 요청들 해제
   557→   557→        pendingRequests.forEach(resolve => resolve());
   558→   558→        pendingRequests = [];
   559→   559→
   560→   560→        // 원래 요청 재시도
   561→   561→        return adminFetch<T>(url, options);
   562→   562→      } finally {
   563→   563→        isRefreshing = false;
   564→   564→      }
   565→   565→    }
   566→   566→  }
   567→   567→
   568→   568→  if (!response.ok) {
   569→   569→    throw new AdminApiError(response.status, body.error);
   570→   570→  }
   571→   571→
   572→   572→  return body.data;
   573→   573→}
   574→   574→```

   713→   713→#### 3.2.1 수정 의사코드
   714→   714→
   715→   715→```typescript
   716→   716→// app/api/admin/cms/[key]/route.ts
   717→   717→const CmsUpdateSchema = z.object({
   718→   718→  value: z.string().min(1, '값은 비어있을 수 없습니다.'),
   719→   719→});
   720→   720→
   721→   721→async function handleUpdateCms(
   722→   722→  ctx: AdminHandlerContext,
   723→   723→  key: string,
   724→   724→  input: z.infer<typeof CmsUpdateSchema>
   725→   725→): Promise<NextResponse> {
   726→   726→  // 타입별 값 검증
   727→   727→  const existing = await prisma.cmsContent.findUnique({ where: { key } });
   728→   728→  if (!existing) {
   729→   729→    return errorResponse(404, 'NOT_FOUND', `CMS 항목 '${key}'이(가) 존재하지 않습니다.`);
   730→   730→  }
   731→   731→
   732→   732→  // 타입별 추가 검증
   733→   733→  const validatedValue = validateCmsValue(existing.type, input.value);
   734→   734→  //   text   → z.string().max(getMaxLength(key))
   735→   735→  //   color  → /^#[0-9A-Fa-f]{6}$/
   736→   736→  //   number → z.number().min(min).max(max)
   737→   737→  //   boolean→ z.boolean()
   738→   738→  //   image  → URL 형식 + 파일 존재 확인
   739→   739→  //   json   → JSON.parse() 성공 여부
   740→   740→
   741→   741→  // 기존 값 저장 (감사 로그용)
   742→   742→  const oldValue = existing.value;
   743→   743→
   744→   744→  // DB 업데이트
   745→   745→  const updated = await prisma.cmsContent.update({
   746→   746→    where: { key },
   747→   747→    data: {
   748→   748→      value: String(validatedValue),
   749→   749→      updatedBy: ctx.admin.id,
   750→   750→      // updatedAt은 @updatedAt으로 자동 갱신
   751→   751→    }
   752→   752→  });
   753→   753→
   754→   754→  // 감사 로그
   755→   755→  await writeAuditLog({
   756→   756→    action: 'UPDATE',
   757→   757→    entity: 'CmsContent',
   758→   758→    entityId: updated.id,
   759→   759→    oldValue: JSON.stringify({ key, value: oldValue }),
   760→   760→    newValue: JSON.stringify({ key, value: updated.value }),
   761→   761→    performedBy: ctx.admin.id,
   762→   762→  });
   763→   763→
   764→   764→  return NextResponse.json({ data: updated });
   765→   765→}
   766→   766→
   767→   767→// 타입별 검증 함수
   768→   768→function validateCmsValue(type: string, value: string): unknown {
   769→   769→  switch (type) {
   770→   770→    case 'text':
   771→   771→      if (value.length > 200) throw new ValidationError('텍스트가 너무 깁니다.');
   772→   772→      return value;
   773→   773→    case 'color':
   774→   774→      if (!/^#[0-9A-Fa-f]{6}$/.test(value)) throw new ValidationError('유효한 HEX 색상이 아닙니다.');
   775→   775→      return value;
   776→   776→    case 'number':
   777→   777→      const num = Number(value);
   778→   778→      if (isNaN(num)) throw new ValidationError('숫자가 아닙니다.');
   779→   779→      return value;
   780→   780→    case 'boolean':
   781→   781→      if (!['true', 'false'].includes(value)) throw new ValidationError('boolean 값이 아닙니다.');
   782→   782→      return value;
   783→   783→    case 'image':
   784→   784→      if (!value.startsWith('/uploads/') && !value.startsWith('/images/'))
   785→   785→        throw new ValidationError('유효한 이미지 경로가 아닙니다.');
   786→   786→      return value;
   787→   787→    case 'json':
   788→   788→      try { JSON.parse(value); return value; }
   789→   789→      catch { throw new ValidationError('유효한 JSON이 아닙니다.'); }
   790→   790→    default:
   791→   791→      return value;
   792→   792→  }
   793→   793→}
   794→   794→```

   833→   833→    key: z.string(),
   834→   834→    value: z.string(),
   835→   835→  })).min(1).max(100, '최대 100개 항목까지 일괄 수정 가능합니다.'),
   836→   836→});
   837→   837→
   838→   838→async function handleBatchUpdate(
   839→   839→  ctx: AdminHandlerContext,
   840→   840→  input: z.infer<typeof CmsBatchSchema>
   841→   841→): Promise<NextResponse> {
   842→   842→  // 1. 전체 항목 사전 검증
   843→   843→  const existingItems = await prisma.cmsContent.findMany({
   844→   844→    where: { key: { in: input.items.map(i => i.key) } }
   845→   845→  });
   846→   846→
   847→   847→  const existingMap = new Map(existingItems.map(i => [i.key, i]));
   848→   848→
   849→   849→  // 존재하지 않는 키 확인
   850→   850→  const missingKeys = input.items.filter(i => !existingMap.has(i.key));
   851→   851→  if (missingKeys.length > 0) {
   852→   852→    return errorResponse(400, 'INVALID_KEYS', `존재하지 않는 키: ${missingKeys.map(k => k.key).join(', ')}`);
   853→   853→  }
   854→   854→
   855→   855→  // 타입별 값 검증
   856→   856→  for (const item of input.items) {
   857→   857→    const existing = existingMap.get(item.key)!;
   858→   858→    try {
   859→   859→      validateCmsValue(existing.type, item.value);
   860→   860→    } catch (e) {
   861→   861→      return errorResponse(400, 'VALIDATION_ERROR', `키 '${item.key}'의 값이 유효하지 않습니다: ${e.message}`);
   862→   862→    }
   863→   863→  }
   864→   864→
   865→   865→  // 2. DB 트랜잭션으로 원자적 업데이트
   866→   866→  const results = await prisma.$transaction(async (tx) => {
   867→   867→    const updated: CmsContent[] = [];
   868→   868→
   869→   869→    for (const item of input.items) {
   870→   870→      const existing = existingMap.get(item.key)!;
   871→   871→
   872→   872→      // 업데이트
   873→   873→      const result = await tx.cmsContent.update({
   874→   874→        where: { key: item.key },
   875→   875→        data: {
   876→   876→          value: item.value,
   877→   877→          updatedBy: ctx.admin.id,
   878→   878→        }
   879→   879→      });
   880→   880→      updated.push(result);
   881→   881→
   882→   882→      // 감사 로그
   883→   883→      await tx.auditLog.create({
   884→   884→        data: {
   885→   885→          action: 'UPDATE',
   886→   886→          entity: 'CmsContent',
   887→   887→          entityId: result.id,
   888→   888→          oldValue: JSON.stringify({ key: item.key, value: existing.value }),
   889→   889→          newValue: JSON.stringify({ key: item.key, value: item.value }),
   890→   890→          performedBy: ctx.admin.id,
   891→   891→        }
   892→   892→      });
   893→   893→    }
   894→   894→
   895→   895→    return updated;
   896→   896→  });
   897→   897→
   898→   898→  return NextResponse.json({
   899→   899→    data: { updated: results.length, items: results }
   900→   900→  });
   901→   901→}
   902→   902→```

   904→   904→### 3.4 키오스크 콘텐츠 조회 (공개 API)
   905→   905→
   906→   906→키오스크 프론트엔드에서 호출하는 인증 불필요 API. **ETag 기반 조건부 요청**으로 대역폭을 최적화한다.
   907→   907→
   908→   908→```
   909→   909→GET /api/cms/content
   910→   910→     │
   911→   911→     ▼
   912→   912→┌────────────────────┐
   913→   913→│ If-None-Match 헤더 │
   914→   914→│ 확인 (ETag)        │
   915→   915→└────┬───────────────┘
   916→   916→     │
   917→   917→     ├── ETag 일치 → 304 Not Modified (body 없음)
   918→   918→     │
   919→   919→     ▼ ETag 불일치 (또는 최초 요청)
   920→   920→┌────────────────────┐
   921→   921→│ CmsContent 전체 조회│
   922→   922→│ prisma.cmsContent  │
   923→   923→│   .findMany()      │
   924→   924→└────┬───────────────┘
   925→   925→     │
   926→   926→     ▼
   927→   927→┌────────────────────┐
   928→   928→│ ETag 생성          │
   929→   929→│ hash(max(updatedAt))│
   930→   930→└────┬───────────────┘
   931→   931→     │
   932→   932→     ▼  200 + ETag 헤더 + { items: CmsContent[] }

   936→   936→
   937→   937→```typescript
   938→   938→// app/api/cms/content/route.ts
   939→   939→async function handleKioskContent(request: NextRequest): Promise<NextResponse> {
   940→   940→  // CmsContent 전체 조회
   941→   941→  const items = await prisma.cmsContent.findMany({
   942→   942→    orderBy: { key: 'asc' }
   943→   943→  });
   944→   944→
   945→   945→  // ETag 생성 (모든 항목의 updatedAt 중 최대값의 해시)
   946→   946→  const maxUpdatedAt = Math.max(...items.map(i => i.updatedAt.getTime()));
   947→   947→  const etag = `"${hashString(String(maxUpdatedAt))}"`;
   948→   948→
   949→   949→  // 조건부 요청 확인
   950→   950→  const ifNoneMatch = request.headers.get('If-None-Match');
   951→   951→  if (ifNoneMatch === etag) {
   952→   952→    return new NextResponse(null, {
   953→   953→      status: 304,
   954→   954→      headers: { ETag: etag }
   955→   955→    });
   956→   956→  }
   957→   957→
   958→   958→  // Map 형태로 변환 (키오스크에서 빠르게 조회)
   959→   959→  const contentMap = items.reduce((acc, item) => {
   960→   960→    acc[item.key] = item.value;
   961→   961→    return acc;
   962→   962→  }, {} as Record<string, string>);
   963→   963→
   964→   964→  return NextResponse.json(
   965→   965→    { data: { items, map: contentMap, updatedAt: new Date(maxUpdatedAt).toISOString() } },
   966→   966→    { headers: { ETag: etag, 'Cache-Control': 'max-age=3' } }
   967→   967→  );
   968→   968→}
   969→   969→```

  1143→  1143→#### 4.3.1 이미지 업로드 의사코드
  1144→  1144→
  1145→  1145→```typescript
  1146→  1146→// app/api/admin/cms/images/route.ts (POST)
  1147→  1147→async function handleImageUpload(request: NextRequest, ctx: AdminHandlerContext) {
  1148→  1148→  const formData = await request.formData();
  1149→  1149→  const file = formData.get('file') as File;
  1150→  1150→
  1151→  1151→  // 1. 서버 측 파일 검증
  1152→  1152→  const ALLOWED_MIMES = ['image/jpeg', 'image/png', 'image/svg+xml'];
  1153→  1153→  if (!ALLOWED_MIMES.includes(file.type)) {
  1154→  1154→    return errorResponse(400, 'INVALID_MIME', 'JPG, PNG, SVG 파일만 업로드 가능합니다.');
  1155→  1155→  }
  1156→  1156→  if (file.size > 2 * 1024 * 1024) {
  1157→  1157→    return errorResponse(400, 'FILE_TOO_LARGE', '파일 크기는 2MB 이하여야 합니다.');
  1158→  1158→  }
  1159→  1159→
  1160→  1160→  // 2. Magic byte 검증 (MIME 위조 방지)
  1161→  1161→  const buffer = Buffer.from(await file.arrayBuffer());
  1162→  1162→  if (!verifyMagicByte(buffer, file.type)) {
  1163→  1163→    return errorResponse(400, 'MIME_MISMATCH', '파일 내용이 MIME 타입과 일치하지 않습니다.');
  1164→  1164→  }
  1165→  1165→
  1166→  1166→  // 3. 이미지 처리 (Sharp)
  1167→  1167→  const fileId = randomUUID();
  1168→  1168→  const uploadDir = path.join(process.cwd(), 'public', 'uploads', 'books');
  1169→  1169→  ensureDir(uploadDir);
  1170→  1170→
  1171→  1171→  let outputPath: string;
  1172→  1172→  let width: number;
  1173→  1173→  let height: number;
  1174→  1174→
  1175→  1175→  if (file.type === 'image/svg+xml') {
  1176→  1176→    // SVG: sanitize 후 그대로 저장
  1177→  1177→    const sanitized = sanitizeSvg(buffer.toString('utf-8'));
  1178→  1178→    outputPath = path.join(uploadDir, `${fileId}.svg`);
  1179→  1179→    fs.writeFileSync(outputPath, sanitized);
  1180→  1180→    width = 0; height = 0; // SVG는 치수 생략
  1181→  1181→  } else {
  1182→  1182→    // JPG/PNG: 리사이즈 + WebP 변환
  1183→  1183→    const processed = await sharp(buffer)
  1184→  1184→      .resize(800, 1200, { fit: 'inside', withoutEnlargement: true })
  1185→  1185→      .webp({ quality: 80 })
  1186→  1186→      .toBuffer();
  1187→  1187→
  1188→  1188→    outputPath = path.join(uploadDir, `${fileId}.webp`);
  1189→  1189→    fs.writeFileSync(outputPath, processed);
  1190→  1190→
  1191→  1191→    const meta = await sharp(processed).metadata();
  1192→  1192→    width = meta.width ?? 0;
  1193→  1193→    height = meta.height ?? 0;
  1194→  1194→  }
  1195→  1195→
  1196→  1196→  const url = `/uploads/books/${path.basename(outputPath)}`;
  1197→  1197→
  1198→  1198→  // 4. CmsImage 메타 저장
  1199→  1199→  const imageRecord = await prisma.cmsImage.create({
  1200→  1200→    data: {
  1201→  1201→      id: fileId,
  1202→  1202→      url,
  1203→  1203→      originalName: file.name,
  1204→  1204→      mimeType: file.type,
  1205→  1205→      size: file.size,
  1206→  1206→      width,
  1207→  1207→      height,
  1208→  1208→      uploadedBy: ctx.admin.id,
  1209→  1209→    }
  1210→  1210→  });
  1211→  1211→
  1212→  1212→  // 5. 감사 로그
  1213→  1213→  await writeAuditLog({

  1347→  1347→        pageSize,

  1619→  1619→} as const;
  1620→  1620→
  1621→  1621→type PermissionCode = keyof typeof PERMISSIONS;
  1622→  1622→
  1623→  1623→// 역할별 권한 매핑
  1624→  1624→const ROLE_PERMISSIONS: Record<string, PermissionCode[]> = {
  1625→  1625→  super_admin: Object.keys(PERMISSIONS) as PermissionCode[], // 모든 권한
  1626→  1626→  admin: [
  1627→  1627→    'cms:read', 'cms:write', 'cms:image',
  1628→  1628→    'book:read', 'book:create', 'book:update', 'book:delete', 'book:import',
  1629→  1629→    'user:read', 'user:create', 'user:update', 'user:deactivate',

  1632→  1632→    'notice:read', 'notice:write',

  1635→  1635→    'cms:read',
  1636→  1636→    'book:read',
  1637→  1637→    'user:read',
  1638→  1638→    'stats:read',
  1639→  1639→    'notice:read',
  1640→  1640→  ],
  1641→  1641→};

  1644→  1644→function hasPermission(role: string, required: PermissionCode): boolean {
  1645→  1645→  const permissions = ROLE_PERMISSIONS[role];
  1646→  1646→  if (!permissions) return false;
  1647→  1647→  return permissions.includes(required);
  1648→  1648→}
  1649→  1649→

  1652→  1652→  return required.some(p => hasPermission(role, p));
  1653→  1653→}
  1654→  1654→
  1655→  1655→// 다중 권한 검사 (모두 만족해야 통과)
  1656→  1656→function hasAllPermissions(role: string, required: PermissionCode[]): boolean {
  1657→  1657→  return required.every(p => hasPermission(role, p));

  1689→  1689→

  1701→  1701→### 6.3 자기 수정 가드 (Self-Modification Guard)
  1702→  1702→
  1703→  1703→관리자는 자신의 역할을 변경할 수 없다.
  1704→  1704→
  1705→  1705→```typescript
  1706→  1706→// app/api/admin/admins/[id]/role/route.ts (PATCH)
  1707→  1707→async function handleUpdateRole(
  1708→  1708→  ctx: AdminHandlerContext,
  1709→  1709→  targetId: string,
  1710→  1710→  input: { newRole: string }
  1711→  1711→) {
  1712→  1712→  // 자기 수정 가드: 자신의 역할 변경 금지
  1713→  1713→  if (targetId === ctx.admin.id) {
  1714→  1714→    return errorResponse(403, 'SELF_MODIFICATION', '자신의 역할은 변경할 수 없습니다.');
  1715→  1715→  }
  1716→  1716→
  1717→  1717→  // 역할 계층 검사: 자신보다 높거나 같은 역할의 계정은 수정 불가
  1718→  1718→  const target = await prisma.adminUser.findUnique({ where: { id: targetId } });
  1719→  1719→  if (!target) return errorResponse(404, 'NOT_FOUND', '관리자를 찾을 수 없습니다.');
  1720→  1720→
  1721→  1721→  if (!canManageRole(ctx.admin.role.code, target.role.code)) {
  1722→  1722→    return errorResponse(403, 'ROLE_HIERARCHY', '자신보다 높거나 같은 권한의 계정은 수정할 수 없습니다.');
  1723→  1723→  }
  1724→  1724→
  1725→  1725→  // 새 역할 검증
  1726→  1726→  if (!ROLE_PERMISSIONS[input.newRole]) {
  1727→  1727→    return errorResponse(400, 'INVALID_ROLE', `유효하지 않은 역할: ${input.newRole}`);
  1728→  1728→  }
  1729→  1729→
  1730→  1730→  const oldRole = target.role.code;
  1731→  1731→
  1732→  1732→  await prisma.adminUser.update({
  1733→  1733→    where: { id: targetId },
  1734→  1734→    data: { role: { connect: { code: input.newRole } } }
  1735→  1735→  });
  1736→  1736→
  1737→  1737→  await writeAuditLog({
  1738→  1738→    action: 'UPDATE',

  1740→  1740→    entityId: targetId,
  1741→  1741→    oldValue: JSON.stringify({ role: oldRole }),
  1742→  1742→    newValue: JSON.stringify({ role: input.newRole }),
  1743→  1743→    performedBy: ctx.admin.id,
  1744→  1744→  });
  1745→  1745→
  1746→  1746→  return NextResponse.json({ data: { success: true } });
  1747→  1747→}
  1748→  1748→```

  1837→  1837→### 7.2 Zustand Store + 폴링 의사코드
  1838→  1838→
  1839→  1839→```typescript
  1840→  1840→// store/cms-store.ts
  1841→  1841→import { create } from 'zustand';
  1842→  1842→
  1843→  1843→interface CmsState {
  1844→  1844→  content: Record<string, string>;   // key → value Map
  1845→  1845→  lastUpdatedAt: string | null;
  1846→  1846→  etag: string | null;
  1847→  1847→  isLoading: boolean;
  1848→  1848→  error: Error | null;
  1849→  1849→
  1850→  1850→  // Actions
  1851→  1851→  setContent: (data: Record<string, string>, etag: string, updatedAt: string) => void;
  1852→  1852→  fetchContent: () => Promise<void>;
  1853→  1853→}
  1854→  1854→
  1855→  1855→export const useCmsStore = create<CmsState>((set, get) => ({
  1856→  1856→  content: {},
  1857→  1857→  lastUpdatedAt: null,
  1858→  1858→  etag: null,
  1859→  1859→  isLoading: false,
  1860→  1860→  error: null,
  1861→  1861→
  1862→  1862→  setContent: (data, etag, updatedAt) => set({
  1863→  1863→    content: data,
  1864→  1864→    etag,
  1865→  1865→    lastUpdatedAt: updatedAt,
  1866→  1866→    isLoading: false,
  1867→  1867→    error: null,
  1868→  1868→  }),
  1869→  1869→
  1870→  1870→  fetchContent: async () => {
  1871→  1871→    const { etag } = get();
  1872→  1872→    set({ isLoading: true });
  1873→  1873→
  1874→  1874→    try {
  1875→  1875→      const headers: Record<string, string> = {};
  1876→  1876→      if (etag) headers['If-None-Match'] = etag;
  1877→  1877→
  1878→  1878→      const response = await fetch('/api/cms/content', { headers });
  1879→  1879→
  1880→  1880→      if (response.status === 304) {
  1881→  1881→        // 콘텐츠 변경 없음
  1882→  1882→        set({ isLoading: false });
  1883→  1883→        return;
  1884→  1884→      }
  1885→  1885→
  1886→  1886→      if (!response.ok) throw new Error(`CMS API 실패: ${response.status}`);
  1887→  1887→
  1888→  1888→      const { data } = await response.json();
  1889→  1889→      const newEtag = response.headers.get('ETag');
  1890→  1890→
  1891→  1891→      set({
  1892→  1892→        content: data.map,
  1893→  1893→        etag: newEtag,
  1894→  1894→        lastUpdatedAt: data.updatedAt,
  1895→  1895→        isLoading: false,
  1896→  1896→        error: null,
  1897→  1897→      });
  1898→  1898→    } catch (error) {
  1899→  1899→      set({ isLoading: false, error: error as Error });
  1900→  1900→      // CMS API 실패 → 기본값 사용 (그레이스풀 폴백)
  1901→  1901→      // content가 비어있으면 defaults로 초기화
  1902→  1902→      const { content } = get();
  1903→  1903→      if (Object.keys(content).length === 0) {
  1904→  1904→        const defaults: Record<string, string> = {};
  1905→  1905→        for (const [key, def] of Object.entries(CMS_DEFAULTS)) {
  1906→  1906→          defaults[key] = def.value;
  1907→  1907→        }
  1908→  1908→        set({ content: defaults });
  1909→  1909→      }
  1910→  1910→    }
  1911→  1911→  },
  1912→  1912→}));
  1913→  1913→```

  1974→  1974→### 7.5 콘텐츠 변경 감지 (updatedAt 비교)
  1975→  1975→
  1976→  1976→```typescript
  1977→  1977→// lib/cms-change-detection.ts
  1978→  1978→export function detectContentChanges(
  1979→  1979→  oldContent: Record<string, string>,
  1980→  1980→  newContent: Record<string, string>
  1981→  1981→): Array<{ key: string; oldValue: string; newValue: string }> {
  1982→  1982→  const changes: Array<{ key: string; oldValue: string; newValue: string }> = [];
  1983→  1983→
  1984→  1984→  for (const key of Object.keys(newContent)) {
  1985→  1985→    if (oldContent[key] !== newContent[key]) {
  1986→  1986→      changes.push({
  1987→  1987→        key,
  1988→  1988→        oldValue: oldContent[key] ?? '(없음)',
  1989→  1989→        newValue: newContent[key],
  1990→  1990→      });
  1991→  1991→    }
  1992→  1992→  }
