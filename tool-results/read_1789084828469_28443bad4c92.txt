   159→   159→## 2. 인증 설계
   160→   160→
   161→   161→### 2.1 인증 아키텍처 개요
   162→   162→
   163→   163→관리자 대시보드는 **JWT 기반 stateless 인증**을 사용한다. Access Token은 단기 수명(15분)으로 API 요청 시 `Authorization` 헤더에 전달하고, Refresh Token은 장기 수명(7일)으로 `httpOnly` 쿠키에 저장하여 자동 갱신한다.
   164→   164→
   165→   165→```
   166→   166→┌──────────┐  POST /api/auth/login  ┌───────────────────┐
   167→   167→│  Login   │ ──────────────────────►│  bcrypt.compare()  │
   168→   168→│  Page    │                        │  ↓                  │
   169→   169→│          │ ◄────────────────────  │  signAccessToken()  │
   170→   170→│          │  Set-Cookie: refresh   │  signRefreshToken() │
   171→   171→└──────────┘                        └───────────────────┘
   172→   172→     │
   173→   173→     ▼  모든 /api/admin/* 요청
   174→   174→┌──────────────────────────────────────────────────────────┐
   175→   175→│  Authorization: Bearer <accessToken>                     │
   176→   176→│  → verifyJwt() → decode role → checkPermission() → 401/403/next │
   177→   177→└──────────────────────────────────────────────────────────┘
   178→   178→     │ accessToken 만료 (15분)
   179→   179→     ▼
   180→   180→┌──────────┐  POST /api/auth/refresh  ┌───────────────────┐
   181→   181→│  Client  │ ────────────────────────►│  verifyRefresh()   │
   182→   182→│  (자동)  │  Cookie: refreshToken    │  ↓                  │
   183→   183→│          │ ◄──────────────────────  │  signAccessToken()  │
   184→   184→│          │  { accessToken }         │  rotateRefresh()    │
   185→   185→└──────────┘                          └───────────────────┘
   186→   186→```

   248→   248→### 2.3 비밀번호 해시 설계
   249→   249→
   250→   250→```typescript
   251→   251→// src/lib/admin/auth.ts
   252→   252→
   253→   253→import bcrypt from 'bcrypt';
   254→   254→
   255→   255→const SALT_ROUNDS = 12;
   256→   256→
   257→   257→/** 비밀번호 해시 생성 (관리자 계정 생성/수정 시) */
   258→   258→export async function hashPassword(plainPassword: string): Promise<string> {
   259→   259→  return bcrypt.hash(plainPassword, SALT_ROUNDS);
   260→   260→}
   261→   261→
   262→   262→/** 비밀번호 검증 (로그인 시) */
   263→   263→export async function verifyPassword(
   264→   264→  plainPassword: string,
   265→   265→  hashedPassword: string
   266→   266→): Promise<boolean> {
   267→   267→  return bcrypt.compare(plainPassword, hashedPassword);
   268→   268→}
   269→   269→
   270→   270→/** 비밀번호 정책 검증 */
   271→   271→export function validatePasswordPolicy(password: string): {
   272→   272→  valid: boolean;
   273→   273→  errors: string[];
   274→   274→} {
   275→   275→  const errors: string[] = [];
   276→   276→
   277→   277→  if (password.length < 8) {
   278→   278→    errors.push('비밀번호는 8자 이상이어야 합니다.');
   279→   279→  }
   280→   280→  if (!/[a-z]/.test(password)) {
   281→   281→    errors.push('소문자를 포함해야 합니다.');
   282→   282→  }
   283→   283→  if (!/[A-Z]/.test(password)) {
   284→   284→    errors.push('대문자를 포함해야 합니다.');
   285→   285→  }
   286→   286→  if (!/[0-9]/.test(password)) {
   287→   287→    errors.push('숫자를 포함해야 합니다.');
   288→   288→  }
   289→   289→  if (!/[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(password)) {
   290→   290→    errors.push('특수문자를 포함해야 합니다.');
   291→   291→  }
   292→   292→
   293→   293→  return { valid: errors.length === 0, errors };
   294→   294→}
   295→   295→```

   297→   297→### 2.4 JWT 발급 및 검증
   298→   298→
   299→   299→```typescript
   300→   300→// src/lib/admin/auth.ts (계속)
   301→   301→
   302→   302→import { SignJWT, jwtVerify } from 'jose';
   303→   303→
   304→   304→const ACCESS_TOKEN_SECRET = new TextEncoder().encode(
   305→   305→  process.env.JWT_ACCESS_SECRET!  // 최소 32바이트 랜덤 문자열
   306→   306→);
   307→   307→const REFRESH_TOKEN_SECRET = new TextEncoder().encode(
   308→   308→  process.env.JWT_REFRESH_SECRET! // 최소 32바이트 랜덤 문자열
   309→   309→);
   310→   310→
   311→   311→const ACCESS_TOKEN_TTL = '15m';   // 15분
   312→   312→const REFRESH_TOKEN_TTL = '7d';   // 7일
   313→   313→
   314→   314→/** Access Token 발급 */
   315→   315→export async function signAccessToken(
   316→   316→  payload: Omit<AccessTokenPayload, 'iat' | 'exp'>
   317→   317→): Promise<string> {
   318→   318→  return new SignJWT({
   319→   319→    sub: payload.sub,
   320→   320→    email: payload.email,
   321→   321→    role: payload.role,
   322→   322→    permissions: payload.permissions,
   323→   323→  })
   324→   324→    .setProtectedHeader({ alg: 'HS256' })
   325→   325→    .setIssuedAt()
   326→   326→    .setExpirationTime(ACCESS_TOKEN_TTL)
   327→   327→    .setIssuer('smart-library-admin')
   328→   328→    .setAudience('admin-api')
   329→   329→    .sign(ACCESS_TOKEN_SECRET);
   330→   330→}
   331→   331→
   332→   332→/** Refresh Token 발급 */
   333→   333→export async function signRefreshToken(
   334→   334→  payload: Omit<RefreshTokenPayload, 'iat' | 'exp'>
   335→   335→): Promise<string> {
   336→   336→  return new SignJWT({
   337→   337→    sub: payload.sub,
   338→   338→    tokenVersion: payload.tokenVersion,
   339→   339→  })
   340→   340→    .setProtectedHeader({ alg: 'HS256' })
   341→   341→    .setIssuedAt()
   342→   342→    .setExpirationTime(REFRESH_TOKEN_TTL)
   343→   343→    .setIssuer('smart-library-admin')
   344→   344→    .setAudience('admin-refresh')
   345→   345→    .sign(REFRESH_TOKEN_SECRET);
   346→   346→}
   347→   347→
   348→   348→/** Access Token 검증 */
   349→   349→export async function verifyAccessToken(
   350→   350→  token: string
   351→   351→): Promise<AccessTokenPayload> {
   352→   352→  const { payload } = await jwtVerify(token, ACCESS_TOKEN_SECRET, {
   353→   353→    issuer: 'smart-library-admin',
   354→   354→    audience: 'admin-api',
   355→   355→  });
   356→   356→  return payload as unknown as AccessTokenPayload;
   357→   357→}
   358→   358→
   359→   359→/** Refresh Token 검증 */
   360→   360→export async function verifyRefreshToken(
   361→   361→  token: string
   362→   362→): Promise<RefreshTokenPayload> {
   363→   363→  const { payload } = await jwtVerify(token, REFRESH_TOKEN_SECRET, {
   364→   364→    issuer: 'smart-library-admin',
   365→   365→    audience: 'admin-refresh',
   366→   366→  });
   367→   367→  return payload as unknown as RefreshTokenPayload;
   368→   368→}

   395→   395→### 2.6 Token 갱신 흐름
   396→   396→
   397→   397→```typescript
   398→   398→// src/app/api/auth/refresh/route.ts
   399→   399→
   400→   400→import { NextRequest, NextResponse } from 'next/server';
   401→   401→import {
   402→   402→  verifyRefreshToken,
   403→   403→  signAccessToken,
   404→   404→  signRefreshToken,
   405→   405→  setRefreshTokenCookie,
   406→   406→  clearRefreshTokenCookie,
   407→   407→} from '@/lib/admin/auth';
   408→   408→import { prisma } from '@/lib/db';
   409→   409→
   410→   410→export async function POST(request: NextRequest) {
   411→   411→  try {
   412→   412→    // 1. 쿠키에서 Refresh Token 추출
   413→   413→    const refreshToken = request.cookies.get('refresh_token')?.value;
   414→   414→    if (!refreshToken) {
   415→   415→      return NextResponse.json(
   416→   416→        { error: { code: 'REFRESH_TOKEN_INVALID', message: 'Refresh token이 없습니다.' } },
   417→   417→        { status: 401 }
   418→   418→      );
   419→   419→    }
   420→   420→
   421→   421→    // 2. Refresh Token 검증
   422→   422→    const payload = await verifyRefreshToken(refreshToken);
   423→   423→
   424→   424→    // 3. DB에서 관리자 정보 조회 (토큰 버전 확인)
   425→   425→    const admin = await prisma.adminUser.findUnique({
   426→   426→      where: { id: payload.sub },
   427→   427→    });
   428→   428→
   429→   429→    if (!admin || !admin.isActive) {
   430→   430→      clearRefreshTokenCookie();
   431→   431→      return NextResponse.json(
   432→   432→        { error: { code: 'ACCOUNT_INACTIVE', message: '비활성화된 계정입니다.' } },
   433→   433→        { status: 401 }
   434→   434→      );
   435→   435→    }
   436→   436→
   437→   437→    // 4. 토큰 버전 불일치 → 강제 로그아웃
   438→   438→    if (admin.tokenVersion !== payload.tokenVersion) {
   439→   439→      clearRefreshTokenCookie();
   440→   440→      return NextResponse.json(
   441→   441→        { error: { code: 'REFRESH_TOKEN_INVALID', message: '토큰이 무효화되었습니다.' } },
   442→   442→        { status: 401 }
   443→   443→      );
   444→   444→    }
   445→   445→
   446→   446→    // 5. 새 Access Token 발급
   447→   447→    const permissions = getPermissionsForRole(admin.role);
   448→   448→    const accessToken = await signAccessToken({
   449→   449→      sub: admin.id,
   450→   450→      email: admin.email,
   451→   451→      role: admin.role as AdminRole,
   452→   452→      permissions,
   453→   453→    });
   454→   454→
   455→   455→    // 6. Refresh Token Rotation (새 Refresh Token 발급)
   456→   456→    const newRefreshToken = await signRefreshToken({
   457→   457→      sub: admin.id,
   458→   458→      tokenVersion: admin.tokenVersion,
   459→   459→    });
   460→   460→    setRefreshTokenCookie(newRefreshToken);
   461→   461→
   462→   462→    // 7. 응답
   463→   463→    return NextResponse.json({
   464→   464→      accessToken,
   465→   465→      user: {
   466→   466→        id: admin.id,
   467→   467→        email: admin.email,
   468→   468→        name: admin.name,
   469→   469→        role: admin.role,
   470→   470→        permissions,
   471→   471→      },
   472→   472→    });
   473→   473→  } catch {
   474→   474→    clearRefreshTokenCookie();
   475→   475→    return NextResponse.json(
   476→   476→      { error: { code: 'TOKEN_INVALID', message: '토큰 검증에 실패했습니다.' } },
   477→   477→      { status: 401 }
   478→   478→    );
   479→   479→  }
   480→   480→}
   481→   481→```

   544→   544→// ─── API Route 보호 미들웨어 ─────────────────────────────
   545→   545→
   546→   546→type HandlerFn = (request: NextRequest, context: any) => Promise<NextResponse>;
   547→   547→
   548→   548→/** 관리자 인증 + 권한 검사 래퍼 */
   549→   549→export function withAdminAuth(
   550→   550→  handler: HandlerFn,
   551→   551→  requiredPermission: Permission
   552→   552→): HandlerFn {
   553→   553→  return async (request: NextRequest, context: any) => {
   554→   554→    try {
   555→   555→      // 1. Authorization 헤더에서 Bearer 토큰 추출
   556→   556→      const authHeader = request.headers.get('Authorization');
   557→   557→      if (!authHeader?.startsWith('Bearer ')) {
   558→   558→        return NextResponse.json(
   559→   559→          { error: { code: 'TOKEN_INVALID', message: '인증 토큰이 필요합니다.' } },
   560→   560→          { status: 401 }
   561→   561→        );
   562→   562→      }
   563→   563→
   564→   564→      const token = authHeader.slice(7);
   565→   565→
   566→   566→      // 2. Access Token 검증
   567→   567→      const payload: AccessTokenPayload = await verifyAccessToken(token);
   568→   568→
   569→   569→      // 3. 권한 확인
   570→   570→      if (!payload.permissions.includes(requiredPermission)) {
   571→   571→        return NextResponse.json(
   572→   572→          { error: { code: 'PERMISSION_DENIED', message: '접근 권한이 없습니다.' } },
   573→   573→          { status: 403 }
   574→   574→        );
   575→   575→      }
   576→   576→
   577→   577→      // 4. 핸들러 실행 (페이로드를 context에 주입)
   578→   578→      return handler(request, { ...context, auth: payload });
   579→   579→    } catch (error) {
   580→   580→      // 토큰 만료 또는 무효
   581→   581→      if (error instanceof Error && error.name === 'JWTExpired') {
   582→   582→        return NextResponse.json(
   583→   583→          { error: { code: 'TOKEN_EXPIRED', message: '액세스 토큰이 만료되었습니다.' } },
   584→   584→          { status: 401 }
   585→   585→        );
   586→   586→      }
   587→   587→      return NextResponse.json(
   588→   588→        { error: { code: 'TOKEN_INVALID', message: '토큰 검증에 실패했습니다.' } },
   589→   589→        { status: 401 }
   590→   590→      );
   591→   591→    }
   592→   592→  };
   593→   593→}

   655→   655→### 2.9 클라이언트 자동 Token 갱신 인터셉터
   656→   656→
   657→   657→```typescript
   658→   658→// src/lib/admin/api-client.ts
   659→   659→
   660→   660→let refreshPromise: Promise<string> | null = null;
   661→   661→
   662→   662→/** 만료된 Access Token을 자동 갱신하는 fetch 래퍼 */
   663→   663→export async function adminFetch<T>(
   664→   664→  url: string,
   665→   665→  options: RequestInit = {}
   666→   666→): Promise<T> {
   667→   667→  const { accessToken, setAccessToken } = useAuthStore.getState();
   668→   668→
   669→   669→  const headers = new Headers(options.headers);
   670→   670→  headers.set('Authorization', `Bearer ${accessToken}`);
   671→   671→
   672→   672→  let response = await fetch(url, { ...options, headers });
   673→   673→
   674→   674→  // 401 → 자동 갱신 시도
   675→   675→  if (response.status === 401) {
   676→   676→    const body = await response.json();
   677→   677→    if (body.error?.code === 'TOKEN_EXPIRED') {
   678→   678→      // 동시 요청 시 갱신 중복 방지
   679→   679→      const newToken = await (refreshPromise ??= refreshToken());
   680→   680→      refreshPromise = null;
   681→   681→
   682→   682→      headers.set('Authorization', `Bearer ${newToken}`);
   683→   683→      response = await fetch(url, { ...options, headers });
   684→   684→    }
   685→   685→  }
   686→   686→
   687→   687→  if (!response.ok) {
   688→   688→    const error = await response.json();
   689→   689→    throw new AdminApiError(response.status, error.error?.code, error.error?.message);
   690→   690→  }
   691→   691→
   692→   692→  return response.json();
   693→   693→}
   694→   694→
   695→   695→/** Refresh Token으로 새 Access Token 발급 */
   696→   696→async function refreshToken(): Promise<string> {
   697→   697→  const res = await fetch('/api/auth/refresh', { method: 'POST' });
   698→   698→  if (!res.ok) {
   699→   699→    useAuthStore.getState().logout();
   700→   700→    window.location.href = '/admin/login';
   701→   701→    throw new Error('인증이 만료되었습니다. 다시 로그인해주세요.');
   702→   702→  }
   703→   703→  const data = await res.json();
   704→   704→  useAuthStore.getState().setAccessToken(data.accessToken);
   705→   705→  return data.accessToken;
   706→   706→}
   707→   707→
   708→   708→/** 관리자 API 에러 클래스 */
   709→   709→export class AdminApiError extends Error {
   710→   710→  constructor(
   711→   711→    public status: number,
   712→   712→    public code: string,
   713→   713→    message: string
   714→   714→  ) {
   715→   715→    super(message);
   716→   716→    this.name = 'AdminApiError';
   717→   717→  }
   718→   718→}
   719→   719→```

   725→   725→### 3.1 Key-Value 콘텐츠 관리 아키텍처
   726→   726→
   727→   727→CMS 콘텐츠는 **Key-Value 패턴**으로 관리한다. 각 화면의 텍스트·이미지·색상·규칙이 `CmsContent` 테이블의 개별 레코드로 저장되며, `key` 값(예: `"idle.title_text"`)으로 프론트엔드가 조회한다. DB에 값이 없으면 코드에 정의된 기본값(`constants.ts`)을 사용한다.

   729→   729→```
   730→   730→┌────────────────────┐         ┌──────────────────────────────┐
   731→   731→│  constants.ts      │         │  CmsContent (SQLite)          │
   732→   732→│  (코드 기본값)      │         │  key → value (JSON string)    │
   733→   733→│                    │         │                                │
   734→   734→│  idle.title:       │         │  "idle.title_text" → "도서관"  │
   735→   735→│    "스마트 도서관"  │         │  "menu.borrow_button_color"   │
   736→   736→│                    │         │    → "#2563EB"                 │
   737→   737→└────────────────────┘         └──────────────────────────────┘
   738→   738→         │                                  │
   739→   739→         │  DB 값 없으면 기본값 사용          │  DB 값 있으면 DB 값 우선
   740→   740→         ▼                                  ▼
   741→   741→┌──────────────────────────────────────────────────────────────┐
   742→   742→│  useCmsContent("idle.title_text", "스마트 도서관")             │
   743→   743→│  → "도서관"  (DB 오버라이드)                                   │
   744→   744→│                                                                │
   745→   745→│  useCmsContent("idle.subtitle_text", "도서 대출·반납 키오스크")  │
   746→   746→│  → "도서 대출·반납 키오스크"  (코드 기본값, DB 오버라이드 없음)  │

  1054→  1054→## 4. 실시간 동기화 설계
  1055→  1055→
  1056→  1056→### 4.1 키오스크 폴링 메커니즘 개요
  1057→  1057→
  1058→  1058→키오스크 프론트엔드는 **3초 간격 폴링**으로 CMS 콘텐츠 변경을 감지한다. 서버는 `ETag` 헤더로 콘텐츠 버전을 관리하고, 클라이언트는 `If-None-Match` 헤더로 조건부 요청을 보내어 변경이 없을 경우 `304 Not Modified`를 수신하여 트래픽을 최소화한다.
  1059→  1059→
  1060→  1060→```
  1061→  1061→┌──────────────┐  GET /api/cms/content        ┌──────────────┐
  1062→  1062→│  Kiosk       │  If-None-Match: "etag_v42"   │  Server      │
  1063→  1063→│  Frontend    │ ────────────────────────────► │  (Next.js)   │
  1064→  1064→│              │                               │              │
  1065→  1065→│              │ ◄── 200 + ETag: "etag_v43"   │  (변경 있음)  │
  1066→  1066→│              │     { content: {...} }        │              │
  1067→  1067→│              │                               │              │
  1068→  1068→│  (3초 후)    │  GET /api/cms/content        │              │
  1069→  1069→│              │  If-None-Match: "etag_v43"   │              │
  1070→  1070→│              │ ────────────────────────────► │              │
  1071→  1071→│              │                               │              │
  1072→  1072→│              │ ◄── 304 Not Modified          │  (변경 없음)  │
  1073→  1073→└──────────────┘                               └──────────────┘
  1074→  1074→```
  1075→  1075→
  1076→  1076→### 4.2 서버: ETag 기반 공개 CMS API
  1077→  1077→
  1078→  1078→```typescript
  1079→  1079→// src/app/api/cms/content/route.ts (키오스크용 공개 API — 인증 불필요)
  1080→  1080→
  1081→  1081→import { NextRequest, NextResponse } from 'next/server';
  1082→  1082→import { prisma } from '@/lib/db';
  1083→  1083→import { CMS_DEFAULTS_MAP, CMS_FIELD_MAP } from '@/lib/admin/cms-defaults';
  1084→  1084→import { createHash } from 'crypto';
  1085→  1085→
  1086→  1086→export async function GET(request: NextRequest) {
  1087→  1087→  // 1. DB에서 모든 CMS 레코드 조회
  1088→  1088→  const dbContents = await prisma.cmsContent.findMany({
  1089→  1089→    orderBy: { key: 'asc' },
  1090→  1090→  });
  1091→  1091→
  1092→  1092→  // 2. 기본값과 병합
  1093→  1093→  const merged: Record<string, string> = {};
  1094→  1094→  for (const [key, defaultValue] of CMS_DEFAULTS_MAP) {
  1095→  1095→    const dbRecord = dbContents.find((c) => c.key === key);
  1096→  1096→    merged[key] = dbRecord?.value ?? defaultValue;
  1097→  1097→  }
  1098→  1098→
  1099→  1099→  // 3. ETag 생성 (내용 해시)
  1100→  1100→  const contentHash = createHash('md5')
  1101→  1101→    .update(JSON.stringify(merged))
  1102→  1102→    .digest('hex');
  1103→  1103→  const etag = `"${contentHash}"`;
  1104→  1104→
  1105→  1105→  // 4. 조건부 요청 검사 (If-None-Match)
  1106→  1106→  const ifNoneMatch = request.headers.get('If-None-Match');
  1107→  1107→  if (ifNoneMatch === etag) {
  1108→  1108→    return new NextResponse(null, {
  1109→  1109→      status: 304,
  1110→  1110→      headers: { ETag: etag },
  1111→  1111→    });
  1112→  1112→  }
  1113→  1113→
  1114→  1114→  // 5. 변경 있음 → 200 + ETag + 콘텐츠
  1115→  1115→  return NextResponse.json(
  1116→  1116→    { content: merged, updatedAt: new Date().toISOString() },
  1117→  1117→    {
  1118→  1118→      headers: {
  1119→  1119→        ETag: etag,
  1120→  1120→        'Cache-Control': 'no-cache',  // 항상 재검증
  1121→  1121→      },
  1122→  1122→    }
  1123→  1123→  );
  1124→  1124→}
  1125→  1125→```
  1126→  1126→

  1144→  1144→ * 연결 실패 시 지수 백오프로 재시도한다.

  1154→  1154→  const poll = useCallback(async () => {
  1155→  1155→    try {
  1156→  1156→      const headers: Record<string, string> = {};
  1157→  1157→      if (etagRef.current) {
  1158→  1158→        headers['If-None-Match'] = etagRef.current;
  1159→  1159→      }
  1160→  1160→
  1161→  1161→      const response = await fetch('/api/cms/content', { headers });
  1162→  1162→
  1163→  1163→      if (response.status === 304) {
  1164→  1164→        // 변경 없음 — 재시도 카운터 리셋
  1165→  1165→        retryCountRef.current = 0;
  1166→  1166→        setConnectionStatus('connected');
  1167→  1167→        return;
  1168→  1168→      }
  1169→  1169→
  1170→  1170→      if (!response.ok) throw new Error(`HTTP ${response.status}`);
  1171→  1171→
  1172→  1172→      // ETag 저장
  1173→  1173→      const newEtag = response.headers.get('ETag');
  1174→  1174→      if (newEtag) etagRef.current = newEtag;
  1175→  1175→
  1176→  1176→      // 콘텐츠 업데이트
  1177→  1177→      const data = await response.json();
  1178→  1178→      setContent(new Map(Object.entries(data.content)));
  1179→  1179→      setLastUpdated(new Date(data.updatedAt));
  1180→  1180→      setConnectionStatus('connected');
  1181→  1181→
  1182→  1182→      // 재시도 카운터 리셋
  1183→  1183→      retryCountRef.current = 0;
  1184→  1184→    } catch (error) {
  1185→  1185→      retryCountRef.current++;
  1186→  1186→      setConnectionStatus('disconnected');
  1187→  1187→
  1188→  1188→      if (retryCountRef.current >= MAX_RETRIES) {
  1189→  1189→        setConnectionStatus('failed');
  1190→  1190→        console.error('[CMS Polling] 최대 재시도 횟수 초과:', error);
  1191→  1191→        return;
  1192→  1192→      }
  1193→  1193→
  1194→  1194→      // 지수 백오프 대기 후 재시도
  1195→  1195→      const delay = RETRY_BASE_DELAY_MS * Math.pow(2, retryCountRef.current - 1);
  1196→  1196→      console.warn(`[CMS Polling] 재시도 ${retryCountRef.current}/${MAX_RETRIES} (${delay}ms 후)`);
  1197→  1197→      await new Promise((r) => setTimeout(r, delay));
  1198→  1198→    }
  1199→  1199→  }, [setContent, setLastUpdated, setConnectionStatus]);
  1200→  1200→
  1201→  1201→  useEffect(() => {
  1202→  1202→    if (!enabled) return;
  1203→  1203→
  1204→  1204→    // 최초 1회 즉시 실행
  1205→  1205→    poll();
  1206→  1206→
  1207→  1207→    // 이후 3초 간격 폴링
  1208→  1208→    intervalRef.current = setInterval(poll, POLLING_INTERVAL_MS);
  1209→  1209→
  1210→  1210→    return () => {
  1211→  1211→      if (intervalRef.current) {
  1212→  1212→        clearInterval(intervalRef.current);
  1213→  1213→      }
  1214→  1214→    };
  1215→  1215→  }, [enabled, poll]);
  1216→  1216→}
  1217→  1217→```

  1439→  1439→### 5.4 매직 넘버 검증 (MIME 스푸핑 방어)
  1440→  1440→
  1441→  1441→```typescript
  1442→  1442→// src/lib/admin/upload.ts (계속)
  1443→  1443→
  1444→  1444→import { IMAGE_MAGIC_NUMBERS, ALLOWED_IMAGE_MIMES, MAX_IMAGE_SIZE } from '@/types/admin';
  1445→  1445→
  1446→  1446→/**
  1447→  1447→ * 파일 매직 넘버 검증
  1448→  1448→ * MIME 헤더만 신뢰하지 않고, 파일의 실제 바이트 시그니처를 확인한다.
  1449→  1449→ */
  1450→  1450→export function validateMagicNumber(buffer: Buffer, claimedMime: string): boolean {
  1451→  1451→  // SVG는 텍스트 포맷 — <svg 태그로 시작하는지 확인
  1452→  1452→  if (claimedMime === 'image/svg+xml') {
  1453→  1453→    const head = buffer.toString('utf8', 0, 256).trim();
  1454→  1454→    return head.startsWith('<?xml') || head.startsWith('<svg');
  1455→  1455→  }
  1456→  1456→
  1457→  1457→  const expected = IMAGE_MAGIC_NUMBERS[claimedMime];
  1458→  1458→  if (!expected || expected.length === 0) return false;
  1459→  1459→
  1460→  1460→  // 파일 시작 바이트와 매직 넘버 비교
  1461→  1461→  for (let i = 0; i < expected.length; i++) {
  1462→  1462→    if (buffer[i] !== expected[i]) return false;
  1463→  1463→  }
  1464→  1464→
  1465→  1465→  // WebP는 RIFF + WEBP 시그니처 확인
  1466→  1466→  if (claimedMime === 'image/webp') {
  1467→  1467→    const webpSig = buffer.toString('ascii', 8, 12);
  1468→  1468→    if (webpSig !== 'WEBP') return false;
  1469→  1469→  }
  1470→  1470→
  1471→  1471→  return true;
  1472→  1472→}
  1473→  1473→
  1474→  1474→/** 이미지 업로드 전체 검증 */
  1475→  1475→export function validateImageUpload(
  1476→  1476→  buffer: Buffer,
  1477→  1477→  claimedMime: string,
  1478→  1478→  fileSize: number
  1479→  1479→): { valid: boolean; error?: string } {
  1480→  1480→  // 1. MIME 타입 허용 여부
  1481→  1481→  if (!ALLOWED_IMAGE_MIMES.includes(claimedMime as any)) {
  1482→  1482→    return { valid: false, error: `허용되지 않는 파일 형식: ${claimedMime}` };
  1483→  1483→  }
  1484→  1484→
  1485→  1485→  // 2. 파일 크기 제한
  1486→  1486→  if (fileSize > MAX_IMAGE_SIZE) {
  1487→  1487→    return { valid: false, error: `파일 크기 초과: ${fileSize} > ${MAX_IMAGE_SIZE} bytes` };
  1488→  1488→  }
  1489→  1489→
  1490→  1490→  // 3. 매직 넘버 검증
  1491→  1491→  if (!validateMagicNumber(buffer, claimedMime)) {
  1492→  1492→    return { valid: false, error: '파일 시그니처가 MIME 타입과 일치하지 않습니다.' };
  1493→  1493→  }
  1494→  1494→
  1495→  1495→  return { valid: true };
  1496→  1496→}
  1497→  1497→```

  1542→  1542→
  1543→  1543→import { NextRequest, NextResponse } from 'next/server';
  1544→  1544→import { withAdminAuth } from '@/lib/admin/rbac';
  1545→  1545→import { validateImageUpload, saveImageFile, generateThumbnail } from '@/lib/admin/upload';
  1546→  1546→import formidable from 'formidable';
  1547→  1547→
  1548→  1548→export const POST = withAdminAuth(async (request, { auth }) => {
  1549→  1549→  // 1. 멀티파트 폼 데이터 파싱
  1550→  1550→  const formData = await request.formData();
  1551→  1551→  const file = formData.get('file') as File | null;
  1552→  1552→  const key = formData.get('key') as string;
  1553→  1553→  const altText = formData.get('altText') as string | null;
  1554→  1554→
  1555→  1555→  if (!file || !key) {
  1556→  1556→    return NextResponse.json(
  1557→  1557→      { error: { code: 'BAD_REQUEST', message: 'file과 key는 필수입니다.' } },
  1558→  1558→      { status: 400 }
  1559→  1559→    );
  1560→  1560→  }
  1561→  1561→
  1562→  1562→  // 2. 파일 버퍼 읽기
  1563→  1563→  const buffer = Buffer.from(await file.arrayBuffer());
  1564→  1564→
  1565→  1565→  // 3. 검증
  1566→  1566→  const validation = validateImageUpload(buffer, file.type, file.size);
  1567→  1567→  if (!validation.valid) {
  1568→  1568→    return NextResponse.json(
  1569→  1569→      { error: { code: 'INVALID_FILE', message: validation.error } },
  1570→  1570→      { status: 400 }
  1571→  1571→    );
  1572→  1572→  }
  1573→  1573→
  1574→  1574→  // 4. 파일 저장
  1575→  1575→  const url = await saveImageFile(buffer, file.name);
  1576→  1576→
  1577→  1577→  // 5. 썸네일 생성
  1578→  1578→  const thumbnailUrl = await generateThumbnail(buffer, file.name);
  1579→  1579→
  1580→  1580→  // 6. CMS 콘텐츠 업데이트
  1581→  1581→  await prisma.cmsContent.upsert({
  1582→  1582→    where: { key },
  1583→  1583→    update: { value: JSON.stringify({ url, thumbnailUrl, altText }), updatedBy: auth.sub },
  1584→  1584→    create: { key, type: 'image', value: JSON.stringify({ url, thumbnailUrl, altText }), updatedBy: auth.sub },
  1585→  1585→  });
  1586→  1586→
  1587→  1587→  // 7. 감사 로그
  1588→  1588→  await recordAuditLog({
  1589→  1589→    action: 'UPDATE',
  1590→  1590→    entity: 'CmsContent',
  1591→  1591→    entityId: key,
  1592→  1592→    newValue: url,
  1593→  1593→    performedBy: auth.sub,
  1594→  1594→  });
  1595→  1595→
  1596→  1596→  return NextResponse.json({
  1597→  1597→    url,
  1598→  1598→    thumbnailUrl: thumbnailUrl || url,  // SVG는 원본 URL
  1599→  1599→    altText,
  1600→  1600→    size: file.size,
  1601→  1601→  });
  1602→  1602→}, 'cms:write');
  1603→  1603→```

  1611→  1611→  thumbnailUrl?: string;

  1618→  1618→export async function updateImageAltText(
