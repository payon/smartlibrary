    89→    89→  approved: 'bg-sky-100 text-sky-700',
    90→    90→  issued: 'bg-emerald-100 text-emerald-700',
    91→    91→  rejected: 'bg-rose-100 text-rose-700',
    92→    92→};
    93→    93→
    94→    94→const CARD_TYPE_LABELS: Record<string, string> = {
    95→    95→  mobile: '모바일',
    96→    96→  physical: '실물카드',
    97→    97→};
    98→    98→
    99→    99→// ============================================================================
   100→   100→// 메인 컴포넌트
   101→   101→// ============================================================================
   102→   102→
   103→   103→export default function CardsSection() {
   104→   104→  // 상태
   105→   105→  const [cards, setCards] = useState<LibraryCard[]>([]);
   106→   106→  const [stats, setStats] = useState<CardStats>({ total: 0, pending: 0, approved: 0, issued: 0, rejected: 0 });
   107→   107→  const [loading, setLoading] = useState(true);
   108→   108→  const [search, setSearch] = useState('');
   109→   109→  const [statusFilter, setStatusFilter] = useState<string>('all');
   110→   110→  const [cardTypeFilter, setCardTypeFilter] = useState<string>('all');
   111→   111→
   112→   112→  // 다이얼로그
   113→   113→  const [detailDialogOpen, setDetailDialogOpen] = useState(false);
   114→   114→  const [issueDialogOpen, setIssueDialogOpen] = useState(false);
   115→   115→  const [rejectDialogOpen, setRejectDialogOpen] = useState(false);
   116→   116→  const [selectedCard, setSelectedCard] = useState<LibraryCard | null>(null);
   117→   117→  const [rejectReason, setRejectReason] = useState('');
   118→   118→  const [saving, setSaving] = useState(false);
   119→   119→
   120→   120→  // 직접 발급 폼
   121→   121→  const [issueForm, setIssueForm] = useState({
   122→   122→    applicantName: '',
   123→   123→    birthDate: '',
   124→   124→    phone: '',
   125→   125→    address: '',
   126→   126→    cardType: 'mobile',
   127→   127→  });
   128→   128→
   129→   129→  // ==========================================================================
   130→   130→  // 데이터 로드
   131→   131→  // ==========================================================================
   132→   132→
   133→   133→  const fetchCards = useCallback(async (searchVal?: string, statusVal?: string, cardTypeVal?: string) => {
   134→   134→    try {
   135→   135→      setLoading(true);
   136→   136→      const params = new URLSearchParams();
   137→   137→      if (searchVal) params.set('search', searchVal);
   138→   138→      if (statusVal && statusVal !== 'all') params.set('status', statusVal);
   139→   139→      if (cardTypeVal && cardTypeVal !== 'all') params.set('cardType', cardTypeVal);
   140→   140→      const res = await fetch(`/api/admin/cards?${params.toString()}`);
   141→   141→      if (!res.ok) throw new Error();
   142→   142→      const json = await res.json();
   143→   143→      setCards(json.cards || []);
   144→   144→      if (json.stats) setStats(json.stats);
   145→   145→    } catch {
   146→   146→      toast.error('도서카드 목록을 불러오는데 실패했습니다.');
   147→   147→    } finally {
   148→   148→      setLoading(false);
   149→   149→    }
   150→   150→  }, []);
   151→   151→
   152→   152→  useEffect(() => {
   153→   153→    fetchCards();
   154→   154→  }, [fetchCards]);
   155→   155→
   156→   156→  // ==========================================================================
   157→   157→  // 액션 핸들러
   158→   158→  // ==========================================================================
   159→   159→
   160→   160→  const handleCardAction = async (cardId: string, action: string, extra?: Record<string, unknown>) => {
   161→   161→    setSaving(true);
   162→   162→    try {
   163→   163→      const res = await fetch(`/api/admin/cards/${cardId}`, {
   164→   164→        method: 'PUT',
   165→   165→        headers: { 'Content-Type': 'application/json' },
   166→   166→        body: JSON.stringify({ action, ...extra }),
   167→   167→      });
   168→   168→      if (!res.ok) {
   169→   169→        const data = await res.json();
   170→   170→        throw new Error(data.error || '작업 실패');
   171→   171→      }
   172→   172→      const data = await res.json();
   173→   173→      toast.success(data.message || '작업이 완료되었습니다.');
   174→   174→      await fetchCards(search, statusFilter, cardTypeFilter);
   175→   175→    } catch (err) {
   176→   176→      toast.error(err instanceof Error ? err.message : '작업에 실패했습니다.');
   177→   177→    } finally {
   178→   178→      setSaving(false);
   179→   179→    }
   180→   180→  };
   181→   181→
   182→   182→  const handleApprove = (card: LibraryCard) => handleCardAction(card.id, 'approve');
   183→   183→  const handleIssue = (card: LibraryCard) => handleCardAction(card.id, 'issue');
   184→   184→
   185→   185→  const handleReject = async () => {
   186→   186→    if (!selectedCard) return;
   187→   187→    await handleCardAction(selectedCard.id, 'reject', { reason: rejectReason });
   188→   188→    setRejectDialogOpen(false);
   189→   189→    setRejectReason('');
   190→   190→    setSelectedCard(null);
   191→   191→  };
   192→   192→
   193→   193→  const handleDelete = async (card: LibraryCard) => {
   194→   194→    if (!confirm(`'${card.applicantName}'님의 도서카드 신청을 삭제하시겠습니까?`)) return;
   195→   195→    try {
   196→   196→      const res = await fetch(`/api/admin/cards/${card.id}`, { method: 'DELETE' });
   197→   197→      if (!res.ok) throw new Error();
   198→   198→      toast.success('도서카드 신청이 삭제되었습니다.');
   199→   199→      await fetchCards(search, statusFilter, cardTypeFilter);
   200→   200→    } catch {
   201→   201→      toast.error('삭제에 실패했습니다.');
   202→   202→    }
   203→   203→  };
   204→   204→
   205→   205→  const handleDirectIssue = async () => {
   206→   206→    const { applicantName, birthDate, phone } = issueForm;
   207→   207→    if (!applicantName || !birthDate || !phone) {
   208→   208→      toast.error('이름, 생년월일, 전화번호는 필수입니다.');
   209→   209→      return;
   210→   210→    }
   211→   211→    if (birthDate.length !== 8) {
   212→   212→      toast.error('생년월일은 8자리로 입력하세요.');
   213→   213→      return;
   214→   214→    }
   215→   215→    setSaving(true);
   216→   216→    try {
   217→   217→      const res = await fetch('/api/admin/cards', {
   218→   218→        method: 'POST',
   219→   219→        headers: { 'Content-Type': 'application/json' },
   220→   220→        body: JSON.stringify(issueForm),
   221→   221→      });
   222→   222→      if (!res.ok) {
   223→   223→        const data = await res.json();
   224→   224→        throw new Error(data.error || '발급 실패');
   225→   225→      }
   226→   226→      const data = await res.json();
   227→   227→      toast.success(`도서카드가 발급되었습니다. (카드번호: ${data.card?.cardNumber})`);
   228→   228→      setIssueDialogOpen(false);
   229→   229→      setIssueForm({ applicantName: '', birthDate: '', phone: '', address: '', cardType: 'mobile' });
   230→   230→      await fetchCards(search, statusFilter, cardTypeFilter);
   231→   231→    } catch (err) {
   232→   232→      toast.error(err instanceof Error ? err.message : '발급에 실패했습니다.');
   233→   233→    } finally {
   234→   234→      setSaving(false);
   235→   235→    }
   236→   236→  };
   237→   237→
   238→   238→  const openDetail = (card: LibraryCard) => {
   239→   239→    setSelectedCard(card);
   240→   240→    setDetailDialogOpen(true);
   241→   241→  };
   242→   242→
   243→   243→  const openReject = (card: LibraryCard) => {
   244→   244→    setSelectedCard(card);
   245→   245→    setRejectDialogOpen(true);
   246→   246→  };
   247→   247→
   248→   248→  // ==========================================================================
   249→   249→  // 렌더
   250→   250→  // ==========================================================================
   251→   251→
   252→   252→  const formatDate = (d: string | null) =>
   253→   253→    d ? new Date(d).toLocaleDateString('ko-KR', { year: 'numeric', month: '2-digit', day: '2-digit' }) : '-';
   254→   254→
   255→   255→  const formatDateTime = (d: string | null) =>
   256→   256→    d ? new Date(d).toLocaleString('ko-KR', { month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit' }) : '-';
   257→   257→
   258→   258→  return (
   259→   259→    <div className="space-y-4">
   260→   260→      {/* ── 통계 카드 ──────────────────────────────────────────────── */}
   261→   261→      <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   262→   262→        <Card className="shadow-sm">
   263→   263→          <CardContent className="p-3 text-center">
   264→   264→            <div className="text-xs text-muted-foreground mb-1">전체</div>
   265→   265→            <div className="text-xl font-bold">{stats.total}</div>
   266→   266→          </CardContent>
   267→   267→        </Card>
   268→   268→        <Card className="shadow-sm border-amber-200">
   269→   269→          <CardContent className="p-3 text-center">
   270→   270→            <div className="text-xs text-amber-600 mb-1 flex items-center justify-center gap-1">
   271→   271→              <Clock className="w-3 h-3" /> 대기
   272→   272→            </div>
   273→   273→            <div className="text-xl font-bold text-amber-700">{stats.pending}</div>
   274→   274→          </CardContent>
   275→   275→        </Card>
   276→   276→        <Card className="shadow-sm border-sky-200">
   277→   277→          <CardContent className="p-3 text-center">
   278→   278→            <div className="text-xs text-sky-600 mb-1 flex items-center justify-center gap-1">
   279→   279→              <CheckCircle2 className="w-3 h-3" /> 승인
   280→   280→            </div>
   281→   281→            <div className="text-xl font-bold text-sky-700">{stats.approved}</div>
   282→   282→          </CardContent>
   283→   283→        </Card>
   284→   284→        <Card className="shadow-sm border-emerald-200">
   285→   285→          <CardContent className="p-3 text-center">
   286→   286→            <div className="text-xs text-emerald-600 mb-1 flex items-center justify-center gap-1">
   287→   287→              <CreditCard className="w-3 h-3" /> 발급완료
   288→   288→            </div>
   289→   289→            <div className="text-xl font-bold text-emerald-700">{stats.issued}</div>
   290→   290→          </CardContent>
   291→   291→        </Card>
   292→   292→        <Card className="shadow-sm border-rose-200 col-span-2 sm:col-span-1">
   293→   293→          <CardContent className="p-3 text-center">
   294→   294→            <div className="text-xs text-rose-600 mb-1 flex items-center justify-center gap-1">
   295→   295→              <XCircle className="w-3 h-3" /> 거부
   296→   296→            </div>
   297→   297→            <div className="text-xl font-bold text-rose-700">{stats.rejected}</div>
   298→   298→          </CardContent>
   299→   299→        </Card>
   300→   300→      </div>
   301→   301→
   302→   302→      {/* ── 필터 & 액션 바 ────────────────────────────────────────── */}
   303→   303→      <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   304→   304→        <div className="relative flex-1 max-w-sm">
   305→   305→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   306→   306→          <Input
   307→   307→            placeholder="이름, 전화번호, 카드번호 검색..."
   308→   308→            value={search}
   309→   309→            onChange={(e) => setSearch(e.target.value)}
   310→   310→            onBlur={() => fetchCards(search, statusFilter, cardTypeFilter)}
   311→   311→            onKeyDown={(e) => e.key === 'Enter' && fetchCards(search, statusFilter, cardTypeFilter)}
   312→   312→            className="pl-9 h-10"
   313→   313→          />
   314→   314→        </div>
   315→   315→        <div className="flex items-center gap-2">
   316→   316→          <Select value={statusFilter} onValueChange={(v) => { setStatusFilter(v); fetchCards(search, v, cardTypeFilter); }}>
   317→   317→            <SelectTrigger className="h-10 w-28">
   318→   318→              <SelectValue placeholder="상태" />
   319→   319→            </SelectTrigger>
   320→   320→            <SelectContent>
   321→   321→              <SelectItem value="all">전체 상태</SelectItem>
   322→   322→              <SelectItem value="pending">대기</SelectItem>
   323→   323→              <SelectItem value="approved">승인</SelectItem>
   324→   324→              <SelectItem value="issued">발급완료</SelectItem>
   325→   325→              <SelectItem value="rejected">거부</SelectItem>
   326→   326→            </SelectContent>
   327→   327→          </Select>
   328→   328→          <Select value={cardTypeFilter} onValueChange={(v) => { setCardTypeFilter(v); fetchCards(search, statusFilter, v); }}>
   329→   329→            <SelectTrigger className="h-10 w-28">
   330→   330→              <SelectValue placeholder="유형" />
   331→   331→            </SelectTrigger>
   332→   332→            <SelectContent>
   333→   333→              <SelectItem value="all">전체 유형</SelectItem>
   334→   334→              <SelectItem value="mobile">모바일</SelectItem>
   335→   335→              <SelectItem value="physical">실물카드</SelectItem>
   336→   336→            </SelectContent>
   337→   337→          </Select>
   338→   338→        </div>
   339→   339→        <div className="flex items-center gap-2 ml-auto">
   340→   340→          {stats.pending > 0 && (
   341→   341→            <Button
   342→   342→              variant="outline"
   343→   343→              size="sm"
   344→   344→              className="h-9 text-amber-700 border-amber-300 hover:bg-amber-50"
   345→   345→              onClick={() => { setStatusFilter('pending'); fetchCards(search, 'pending', cardTypeFilter); }}
   346→   346→            >
   347→   347→              <AlertCircle className="w-4 h-4 mr-1" />
   348→   348→              대기 {stats.pending}건
   349→   349→            </Button>
   350→   350→          )}
   351→   351→          <Button onClick={() => setIssueDialogOpen(true)} className="h-10">
   352→   352→            <Plus className="w-4 h-4 mr-1.5" />
   353→   353→            직접 발급
   354→   354→          </Button>
   355→   355→        </div>
   356→   356→      </div>
   357→   357→
   358→   358→      {/* ── 도서카드 목록 테이블 ───────────────────────────────────── */}
   359→   359→      <Card className="shadow-sm">
   360→   360→        <CardContent className="p-0">
   361→   361→          <div className="overflow-x-auto">
   362→   362→            <Table>
   363→   363→              <TableHeader>
   364→   364→                <TableRow>
   365→   365→                  <TableHead>신청자</TableHead>
   366→   366→                  <TableHead className="hidden md:table-cell">전화번호</TableHead>
   367→   367→                  <TableHead>카드유형</TableHead>
   368→   368→                  <TableHead>카드번호</TableHead>
   369→   369→                  <TableHead>상태</TableHead>
   370→   370→                  <TableHead className="hidden lg:table-cell">신청일</TableHead>
   371→   371→                  <TableHead className="text-center">관리</TableHead>
   372→   372→                </TableRow>
   373→   373→              </TableHeader>
   374→   374→              <TableBody>
   375→   375→                {loading ? (
   376→   376→                  Array.from({ length: 5 }).map((_, i) => (
   377→   377→                    <TableRow key={i}>
   378→   378→                      <TableCell colSpan={7}><Skeleton className="h-8 w-full" /></TableCell>
   379→   379→                    </TableRow>
   380→   380→                  ))
   381→   381→                ) : cards.length === 0 ? (
   382→   382→                  <TableRow>
   383→   383→                    <TableCell colSpan={7} className="text-center py-12 text-muted-foreground">
   384→   384→                      <CreditCard className="w-8 h-8 mx-auto mb-2 opacity-40" />
   385→   385→                      도서카드 발급 신청이 없습니다.
   386→   386→                    </TableCell>
   387→   387→                  </TableRow>
   388→   388→                ) : (
   389→   389→                  cards.map((card) => (
   390→   390→                    <TableRow key={card.id} className="group">
   391→   391→                      <TableCell className="font-medium text-sm">{card.applicantName}</TableCell>
   392→   392→                      <TableCell className="hidden md:table-cell text-sm">{card.phone}</TableCell>
   393→   393→                      <TableCell>
   394→   394→                        <Badge variant="outline" className="text-xs">
   395→   395→                          {CARD_TYPE_LABELS[card.cardType] || card.cardType}
   396→   396→                        </Badge>
   397→   397→                      </TableCell>
   398→   398→                      <TableCell className="text-sm font-mono">
   399→   399→                        {card.cardNumber || '-'}
   400→   400→                      </TableCell>
   401→   401→                      <TableCell>
   402→   402→                        <Badge className={`text-xs ${STATUS_COLORS[card.status] || ''}`}>
   403→   403→                          {STATUS_LABELS[card.status] || card.status}
   404→   404→                        </Badge>
   405→   405→                      </TableCell>
   406→   406→                      <TableCell className="hidden lg:table-cell text-xs text-muted-foreground">
   407→   407→                        {formatDate(card.appliedAt)}
   408→   408→                      </TableCell>
   409→   409→                      <TableCell>
   410→   410→                        <div className="flex items-center justify-center gap-1">
   411→   411→                          {/* 상세 보기 */}
   412→   412→                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => openDetail(card)}>
   413→   413→                            <Eye className="w-4 h-4" />
   414→   414→                          </Button>
   415→   415→
   416→   416→                          {/* 상태별 액션 */}
   417→   417→                          {card.status === 'pending' && (
   418→   418→                            <>
