   181→// IssuanceWorkflowPipeline 서브 컴포넌트
   182→// ============================================================================
   183→
   184→function IssuanceWorkflowPipeline({ stats }: { stats: CardStats }) {
   185→  const steps = [
   186→    {
   187→      id: 'apply',
   188→      label: '신청',
   189→      icon: UserPlus,
   190→      count: stats.pending,
   191→      countLabel: `${stats.pending}건 대기`,
   192→      status: stats.pending > 0 ? 'active' : (stats.total > 0 ? 'completed' : 'pending') as 'active' | 'completed' | 'pending',
   193→    },
   194→    {
   195→      id: 'approve',
   196→      label: '승인',
   197→      icon: CheckCircle2,
   198→      count: stats.approved,
   199→      countLabel: `${stats.approved}건 대기`,
   200→      status: stats.approved > 0 ? 'active' : (stats.issued > 0 ? 'completed' : 'pending') as 'active' | 'completed' | 'pending',
   201→    },
   202→    {
   203→      id: 'issue',
   204→      label: '발급',
   205→      icon: CreditCard,
   206→      count: stats.issued,
   207→      countLabel: `${stats.issued}건 완료`,
   208→      status: stats.issued > 0 ? 'completed' : 'pending' as 'completed' | 'pending',
   209→    },
   210→  ];
   211→
   212→  const statusStyles = {
   213→    completed: 'bg-emerald-500 text-white border-emerald-400 shadow-emerald-500/30',
   214→    active: 'bg-sky-500 text-white border-sky-400 shadow-sky-500/30',
   215→    pending: 'bg-slate-700 text-slate-300 border-slate-600',
   216→  };
   217→
   218→  const lineStyles = {
   219→    completed: 'bg-emerald-500',
   220→    active: 'bg-sky-500',
   221→    pending: 'bg-slate-700',
   222→  };
   223→
   224→  const iconColorStyles = {
   225→    completed: 'text-white',
   226→    active: 'text-white',
   227→    pending: 'text-slate-400',
   228→  };
   229→
   230→  return (
   231→    <Card className="bg-slate-900 border-slate-700 shadow-md">
   232→      <CardContent className="p-5">
   233→        <div className="flex items-center justify-between mb-4">
   234→          <h3 className="text-sm font-semibold text-white flex items-center gap-2">
   235→            <ArrowRight className="w-4 h-4 text-sky-400" />
   236→            발급 워크플로우
   237→          </h3>
   238→          <span className="text-xs text-slate-400">전체 {stats.total}건</span>
   239→        </div>
   240→        <div className="flex items-center justify-center">
   241→          {steps.map((step, idx) => {
   242→            const Icon = step.icon;
   243→            return (
   244→              <div key={step.id} className="flex items-center">
   245→                {/* 스텝 노드 */}
   246→                <div className="flex flex-col items-center gap-2">
   247→                  <div
   248→                    className={`w-14 h-14 rounded-xl border flex items-center justify-center shadow-lg transition-all duration-300 ${statusStyles[step.status]}`}
   249→                  >
   250→                    <Icon className={`w-6 h-6 ${iconColorStyles[step.status]}`} />
   251→                  </div>
   252→                  <span className={`text-xs font-semibold ${step.status === 'pending' ? 'text-slate-400' : 'text-white'}`}>
   253→                    {step.label}
   254→                  </span>
   255→                  <span className={`text-[10px] px-2 py-0.5 rounded-full ${
   256→                    step.status === 'completed' ? 'bg-emerald-500/20 text-emerald-300' :
   257→                    step.status === 'active' ? 'bg-sky-500/20 text-sky-300' :
   258→                    'bg-slate-700 text-slate-400'
   259→                  }`}>
   260→                    {step.countLabel}
   261→                  </span>
   262→                </div>
   263→
   264→                {/* 연결 화살표 (마지막 제외) */}
   265→                {idx < steps.length - 1 && (
   266→                  <div className="flex items-center mx-2 sm:mx-4">
   267→                    <div className={`w-8 sm:w-16 h-0.5 ${lineStyles[steps[idx + 1].status === 'pending' ? 'pending' : 'completed']}`} />
   268→                    <ChevronRight className={`w-4 h-4 -ml-1 ${
   269→                      steps[idx + 1].status === 'pending' ? 'text-slate-500' : 'text-emerald-500'
   270→                    }`} />
   271→                  </div>
   272→                )}
   273→              </div>
   274→            );
   275→          })}
   276→        </div>
   277→      </CardContent>
   278→    </Card>
   279→  );
   280→}

   299→  );

   306→export default function CardsSection() {
   307→  // 상태
   308→  const [cards, setCards] = useState<LibraryCard[]>([]);
   309→  const [stats, setStats] = useState<CardStats>({ total: 0, pending: 0, approved: 0, issued: 0, rejected: 0 });
   310→  const [loading, setLoading] = useState(true);
   311→  const [search, setSearch] = useState('');
   312→  const [statusFilter, setStatusFilter] = useState<string>('all');
   313→  const [cardTypeFilter, setCardTypeFilter] = useState<string>('all');
   314→
   315→  // 다이얼로그
   316→  const [detailDialogOpen, setDetailDialogOpen] = useState(false);
   317→  const [issueDialogOpen, setIssueDialogOpen] = useState(false);
   318→  const [rejectDialogOpen, setRejectDialogOpen] = useState(false);
   319→  const [selectedCard, setSelectedCard] = useState<LibraryCard | null>(null);
   320→  const [rejectReason, setRejectReason] = useState('');
   321→  const [saving, setSaving] = useState(false);
   322→
   323→  // 직접 발급 폼
   324→  const [issueForm, setIssueForm] = useState({
   325→    applicantName: '',
   326→    birthDate: '',
   327→    phone: '',
   328→    address: '',
   329→    cardType: 'mobile',
   330→  });
   331→
   332→  // ==========================================================================
   333→  // 데이터 로드
   334→  // ==========================================================================
   335→
   336→  const fetchCards = useCallback(async (searchVal?: string, statusVal?: string, cardTypeVal?: string) => {
   337→    try {
   338→      setLoading(true);
   339→      const params = new URLSearchParams();
   340→      if (searchVal) params.set('search', searchVal);
   341→      if (statusVal && statusVal !== 'all') params.set('status', statusVal);
   342→      if (cardTypeVal && cardTypeVal !== 'all') params.set('cardType', cardTypeVal);
   343→      const res = await fetch(`/api/admin/cards?${params.toString()}`);
   344→      if (!res.ok) throw new Error();
   345→      const json = await res.json();
   346→      setCards(json.cards || []);
   347→      if (json.stats) setStats(json.stats);
   348→    } catch {
   349→      toast.error('도서카드 목록을 불러오는데 실패했습니다.');
   350→    } finally {
   351→      setLoading(false);
   352→    }
   353→  }, []);
   354→
   355→  useEffect(() => {
   356→    fetchCards();
   357→  }, [fetchCards]);
   358→
   359→  // ==========================================================================
   360→  // 액션 핸들러
   361→  // ==========================================================================
   362→
   363→  const handleCardAction = async (cardId: string, action: string, extra?: Record<string, unknown>) => {
   364→    setSaving(true);
   365→    try {
   366→      const res = await fetch(`/api/admin/cards/${cardId}`, {
   367→        method: 'PUT',
   368→        headers: { 'Content-Type': 'application/json' },
   369→        body: JSON.stringify({ action, ...extra }),
   370→      });
   371→      if (!res.ok) {
   372→        const data = await res.json();
   373→        throw new Error(data.error || '작업 실패');
   374→      }
   375→      const data = await res.json();
   376→      toast.success(data.message || '작업이 완료되었습니다.');
   377→      await fetchCards(search, statusFilter, cardTypeFilter);
   378→    } catch (err) {
   379→      toast.error(err instanceof Error ? err.message : '작업에 실패했습니다.');
   380→    } finally {
   381→      setSaving(false);
   382→    }
   383→  };
   384→
   385→  const handleApprove = (card: LibraryCard) => handleCardAction(card.id, 'approve');
   386→  const handleIssue = (card: LibraryCard) => handleCardAction(card.id, 'issue');
   387→
   388→  const handleReject = async () => {
   389→    if (!selectedCard) return;
   390→    await handleCardAction(selectedCard.id, 'reject', { reason: rejectReason });
   391→    setRejectDialogOpen(false);
   392→    setRejectReason('');
   393→    setSelectedCard(null);
   394→  };
   395→
   396→  const handleDelete = async (card: LibraryCard) => {
   397→    if (!confirm(`'${card.applicantName}'님의 도서카드 신청을 삭제하시겠습니까?`)) return;
   398→    try {
   399→      const res = await fetch(`/api/admin/cards/${card.id}`, { method: 'DELETE' });
   400→      if (!res.ok) throw new Error();
   401→      toast.success('도서카드 신청이 삭제되었습니다.');
   402→      await fetchCards(search, statusFilter, cardTypeFilter);
   403→    } catch {
   404→      toast.error('삭제에 실패했습니다.');
   405→    }
   406→  };
   407→
   408→  const handleDirectIssue = async () => {
   409→    const { applicantName, birthDate, phone } = issueForm;
   410→    if (!applicantName || !birthDate || !phone) {
   411→      toast.error('이름, 생년월일, 전화번호는 필수입니다.');
   412→      return;
   413→    }
   414→    if (birthDate.length !== 8) {
   415→      toast.error('생년월일은 8자리로 입력하세요.');
   416→      return;
   417→    }
   418→    setSaving(true);
   419→    try {
   420→      const res = await fetch('/api/admin/cards', {
   421→        method: 'POST',
   422→        headers: { 'Content-Type': 'application/json' },
   423→        body: JSON.stringify(issueForm),
   424→      });
   425→      if (!res.ok) {
   426→        const data = await res.json();
   427→        throw new Error(data.error || '발급 실패');
   428→      }
   429→      const data = await res.json();
   430→      toast.success(`도서카드가 발급되었습니다. (카드번호: ${data.card?.cardNumber})`);
   431→      setIssueDialogOpen(false);
   432→      setIssueForm({ applicantName: '', birthDate: '', phone: '', address: '', cardType: 'mobile' });
   433→      await fetchCards(search, statusFilter, cardTypeFilter);
   434→    } catch (err) {
   435→      toast.error(err instanceof Error ? err.message : '발급에 실패했습니다.');
   436→    } finally {
   437→      setSaving(false);
   438→    }
   439→  };
   440→
   441→  const openDetail = (card: LibraryCard) => {
   442→    setSelectedCard(card);
   443→    setDetailDialogOpen(true);
   444→  };
   445→
   446→  const openReject = (card: LibraryCard) => {
   447→    setSelectedCard(card);
   448→    setRejectDialogOpen(true);
   449→  };
   450→
   451→  // ==========================================================================
   452→  // 렌더
   453→  // ==========================================================================
   454→
   455→  const formatDate = (d: string | null) =>
   456→    d ? new Date(d).toLocaleDateString('ko-KR', { year: 'numeric', month: '2-digit', day: '2-digit' }) : '-';
   457→
   458→  const formatDateTime = (d: string | null) =>
   459→    d ? new Date(d).toLocaleString('ko-KR', { month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit' }) : '-';
   460→
   461→  return (
   462→    <div className="space-y-5">
   463→      {/* ── 섹션 헤더 (키오스크 스타일) ─────────────────────────────── */}
   464→      <div className="relative">
   465→        <h2 className="text-lg font-bold text-slate-900 dark:text-white flex items-center gap-2">
   466→          <CreditCard className="w-5 h-5 text-sky-500" />
   467→          도서카드 발급 관리
   468→        </h2>
   469→        <div className="mt-1.5 h-[3px] w-32 rounded-full" style={{ background: 'linear-gradient(90deg, #0ea5e9, #38bdf8, transparent)' }} />
   470→      </div>
   471→
   472→      {/* ── 발급 워크플로우 파이프라인 ─────────────────────────────── */}
   473→      <IssuanceWorkflowPipeline stats={stats} />
   474→
   475→      {/* ── 통계 카드 (키오스크 그라디언트 스타일) ────────────────────── */}
   476→      <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   477→        {/* 전체 */}
   478→        <div className="relative overflow-hidden rounded-xl p-4 text-white shadow-md" style={{ background: 'linear-gradient(135deg, #475569, #334155)' }}>
   479→          <div className="absolute top-0 right-0 w-20 h-20 rounded-full bg-white/5 -translate-y-6 translate-x-6" />
   480→          <div className="relative z-10">
