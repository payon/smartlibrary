    68→/** Category badge colors using kiosk accent palette */
    69→const CATEGORY_COLORS: Record<string, string> = {
    70→  소설: 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30',
    71→  인문: 'bg-sky-500/20 text-sky-400 border-sky-500/30',
    72→  과학: 'bg-amber-500/20 text-amber-400 border-amber-500/30',
    73→  역사: 'bg-violet-500/20 text-violet-400 border-violet-500/30',
    74→  시: 'bg-rose-500/20 text-rose-400 border-rose-500/30',
    75→};

    77→const getCategoryBadgeClass = (category: string | null) => {
    78→  if (!category) return 'bg-slate-500/20 text-slate-400 border-slate-500/30';
    79→  return CATEGORY_COLORS[category] || 'bg-slate-500/20 text-slate-400 border-slate-500/30';
    80→};

    95→/** Book cover component with fallback */
    96→function BookCover({
    97→  coverUrl,
    98→  title,
    99→  size = 'table',
   100→}: {
   101→  coverUrl: string | null;
   102→  title: string;
   103→  size?: 'table' | 'grid';
   104→}) {
   105→  if (size === 'grid') {
   106→    if (coverUrl) {
   107→      return (
   108→        <div className="w-full aspect-[3/4] overflow-hidden bg-slate-800">
   109→          <img src={coverUrl} alt={title} className="w-full h-full object-cover" />
   110→        </div>
   111→      );
   112→    }
   113→    return (
   114→      <div className="w-full aspect-[3/4] bg-gradient-to-br from-slate-800 to-slate-900 flex items-center justify-center">
   115→        <BookOpen className="w-8 h-8 text-slate-600" />
   116→      </div>
   117→    );
   118→  }
   119→
   120→  // table size
   121→  if (coverUrl) {
   122→    return (
   123→      <div className="w-12 h-16 rounded overflow-hidden bg-slate-800">
   124→        <img src={coverUrl} alt={title} className="w-full h-full object-cover" />
   125→      </div>
   126→    );
   127→  }
   128→  return (
   129→    <div className="w-12 h-16 rounded bg-gradient-to-br from-slate-800 to-slate-900 flex items-center justify-center">
   130→      <BookOpen className="w-5 h-5 text-slate-600" />
   131→    </div>
   132→  );
   133→}

   135→export default function BooksSection() {
   136→  const [books, setBooks] = useState<Book[]>([]);
   137→  const [loading, setLoading] = useState(true);
   138→  const [search, setSearch] = useState('');
   139→  const [category, setCategory] = useState('전체');
   140→  const [viewMode, setViewMode] = useState<'table' | 'grid'>('table');
   141→  const [editDialogOpen, setEditDialogOpen] = useState(false);
   142→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   143→  const [selectedBook, setSelectedBook] = useState<Book | null>(null);
   144→  const [form, setForm] = useState(emptyBook);
   145→  const [isAdding, setIsAdding] = useState(false);
   146→  const [saving, setSaving] = useState(false);
   147→
   148→  const fetchBooks = useCallback(async () => {
   149→    try {
   150→      setLoading(true);
   151→      const params = new URLSearchParams();
   152→      if (search) params.set('search', search);
   153→      if (category && category !== '전체') params.set('category', category);
   154→      const res = await fetch(`/api/admin/books?${params.toString()}`);
   155→      if (!res.ok) throw new Error();

   157→      setBooks(json.books || []);
   158→    } catch {
   159→      toast.error('도서 목록을 불러오는데 실패했습니다.');
   160→    } finally {
   161→      setLoading(false);
   162→    }
   163→  }, [search, category]);
   164→
   165→  useEffect(() => {
   166→    fetchBooks();
   167→  }, [fetchBooks]);
   168→
   169→  const handleAdd = () => {
   170→    setIsAdding(true);
   171→    setForm(emptyBook);
   172→    setSelectedBook(null);
   173→    setEditDialogOpen(true);
   174→  };
   175→
   176→  const handleEdit = (book: Book) => {
   177→    setIsAdding(false);
   178→    setSelectedBook(book);
   179→    setForm({
   180→      isbn: book.isbn,
   181→      title: book.title,
   182→      author: book.author,
   183→      publisher: book.publisher || '',
   184→      publishYear: book.publishYear,
   185→      category: book.category || '',
   186→      coverUrl: book.coverUrl || '',
   187→      totalCopies: book.totalCopies,
   188→      availableCopies: book.availableCopies,
   189→      shelfLocation: book.shelfLocation || '',

   192→  };
   193→
   194→  const handleDelete = (book: Book) => {
   195→    setSelectedBook(book);
   196→    setDeleteDialogOpen(true);
   197→  };
   198→
   199→  const handleSave = async () => {

   201→      toast.error('ISBN, 제목, 저자는 필수입니다.');
   202→      return;
   203→    }
   204→
   205→    setSaving(true);
   206→    try {
   207→      if (isAdding) {
   208→        const res = await fetch('/api/admin/books', {
   209→          method: 'POST',
   210→          headers: { 'Content-Type': 'application/json' },
   211→          body: JSON.stringify(form),
   212→        });
   213→        if (!res.ok) {
   214→          const data = await res.json();
   215→          throw new Error(data.error || '등록 실패');
   216→        }
   217→        toast.success('도서가 등록되었습니다.');
   218→      } else if (selectedBook) {
   219→        const res = await fetch(`/api/admin/books/${selectedBook.id}`, {
   220→          method: 'PUT',
   221→          headers: { 'Content-Type': 'application/json' },
   222→          body: JSON.stringify(form),
   223→        });
   224→        if (!res.ok) {
   225→          const data = await res.json();
   226→          throw new Error(data.error || '수정 실패');
   227→        }
   228→        toast.success('도서가 수정되었습니다.');
   229→      }
   230→      setEditDialogOpen(false);
   231→      await fetchBooks();
   232→    } catch (err) {
   233→      toast.error(err instanceof Error ? err.message : '저장에 실패했습니다.');
   234→    } finally {
   235→      setSaving(false);
   236→    }
   237→  };
   238→
   239→  const handleConfirmDelete = async () => {
   240→    if (!selectedBook) return;
   241→    try {
   242→      const res = await fetch(`/api/admin/books/${selectedBook.id}`, {
   243→        method: 'DELETE',
   244→      });
   245→      if (!res.ok) {
   246→        const data = await res.json();
   247→        throw new Error(data.error || '삭제 실패');
   248→      }
   249→      toast.success('도서가 삭제되었습니다.');
   250→      setDeleteDialogOpen(false);
   251→      await fetchBooks();
   252→    } catch (err) {
   253→      toast.error(err instanceof Error ? err.message : '삭제에 실패했습니다.');
   254→    }
   255→  };
   256→
   257→  const inputBase = 'bg-slate-800 border-slate-700 text-white placeholder:text-slate-500 focus-visible:ring-sky-500';
   258→
   259→  return (
   260→    <div className="space-y-4">
   261→      {/* Section Header with hero banner + gradient underline */}
   262→      <div className="relative overflow-hidden rounded-xl h-[120px]">
   263→        <Image
   264→          src="/images/admin/books-hero.png"
   265→          alt="도서 관리"
   266→          fill
   267→          className="object-cover"
   268→        />

   270→        <div className="absolute inset-0 flex items-center px-6">
   271→          <div>
   272→            <h2 className="text-2xl font-bold text-white">도서 관리</h2>
   273→            <div className="mt-1 h-0.5 w-24 bg-gradient-to-r from-sky-500 via-sky-400 to-transparent" />
   274→            <p className="text-slate-400 text-sm mt-2">도서 등록, 수정, 재고 관리</p>
   275→          </div>
   276→        </div>
   277→      </div>
   278→
   279→      {/* 상단 바: 검색 + 필터 + 뷰 토글 + 추가 */}

   281→        <div className="relative flex-1 min-w-[200px]">
   282→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-500" />
   283→          <Input
   284→            placeholder="제목, 저자, ISBN 검색..."
   285→            value={search}
   286→            onChange={(e) => setSearch(e.target.value)}
   287→            className={`pl-9 h-10 ${inputBase}`}

   289→        </div>

   295→            {CATEGORIES.map((cat) => (

   297→                {cat}
   298→              </SelectItem>
   299→            ))}
   300→          </SelectContent>
   301→        </Select>
   302→        {/* View toggle */}
   303→        <div className="flex items-center bg-slate-900 border border-slate-700 rounded-lg overflow-hidden">
   304→          <button
   305→            onClick={() => setViewMode('table')}
   306→            className={`h-10 w-10 flex items-center justify-center transition-colors ${
   307→              viewMode === 'table'
   308→                ? 'bg-sky-500 text-white'
   309→                : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800'
   310→            }`}

   312→            <List className="w-4 h-4" />

   315→            onClick={() => setViewMode('grid')}
   316→            className={`h-10 w-10 flex items-center justify-center transition-colors ${
   317→              viewMode === 'grid'
   318→                ? 'bg-sky-500 text-white'
   319→                : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800'
   320→            }`}
   321→          >
   322→            <LayoutGrid className="w-4 h-4" />
   323→          </button>
   324→        </div>
   325→        <button
   326→          onClick={handleAdd}
   327→          className="h-10 px-4 text-sm rounded-lg bg-gradient-to-r from-sky-500 to-sky-600 text-white hover:from-sky-400 hover:to-sky-500 transition-all flex items-center gap-1.5 shadow-lg shadow-sky-500/20"
   328→        >
   329→          <Plus className="w-4 h-4" />
   330→          도서 추가
   331→        </button>
   332→      </div>
   333→
   334→      {/* 도서 목록 */}
   335→      {viewMode === 'table' ? (
   336→        /* TABLE VIEW */
   337→        <Card className="bg-slate-900 border-slate-700 shadow-sm">
   338→          <CardContent className="p-0">
   339→            <div className="overflow-x-auto">
   340→              <Table>
   341→                <TableHeader>
   342→                  <TableRow className="bg-slate-800 hover:bg-slate-800 border-slate-700">
   343→                    <TableHead className="w-20 text-slate-400">표지</TableHead>
   344→                    <TableHead className="text-slate-400">제목</TableHead>
   345→                    <TableHead className="hidden md:table-cell text-slate-400">저자</TableHead>
   346→                    <TableHead className="hidden lg:table-cell text-slate-400">카테고리</TableHead>
   347→                    <TableHead className="text-center text-slate-400">재고</TableHead>
   348→                    <TableHead className="text-center text-slate-400">대출가능</TableHead>
   349→                    <TableHead className="text-center w-24 text-slate-400">관리</TableHead>
   350→                  </TableRow>
   351→                </TableHeader>
   352→                <TableBody>
   353→                  {loading ? (
   354→                    Array.from({ length: 5 }).map((_, i) => (
   355→                      <TableRow key={i} className="border-slate-700">
   356→                        <TableCell colSpan={7}>
   357→                          <Skeleton className="h-10 w-full bg-slate-800" />
   358→                        </TableCell>
   359→                      </TableRow>
   360→                    ))
   361→                  ) : books.length === 0 ? (
   362→                    <TableRow className="border-slate-700">
   363→                      <TableCell colSpan={7} className="text-center py-8 text-slate-400">
   364→                        <BookOpen className="w-8 h-8 mx-auto mb-2 text-slate-600" />
   365→                        도서가 없습니다.
   366→                      </TableCell>
   367→                    </TableRow>
   368→                  ) : (
   369→                    books.map((book) => (
   370→                      <TableRow key={book.id} className="border-slate-700 hover:bg-slate-800/60 transition-colors duration-150">
   371→                        <TableCell>
   372→                          <BookCover coverUrl={book.coverUrl} title={book.title} size="table" />
   373→                        </TableCell>
   374→                        <TableCell>
   375→                          <div>
   376→                            <p className="font-medium text-sm text-white">{book.title}</p>
   377→                            <p className="text-xs text-slate-400 md:hidden">{book.author}</p>
   378→                          </div>
   379→                        </TableCell>
   380→                        <TableCell className="hidden md:table-cell text-sm text-slate-300">{book.author}</TableCell>
   381→                        <TableCell className="hidden lg:table-cell">
   382→                          {book.category && (
   383→                            <span className={`text-xs px-2 py-0.5 rounded border ${getCategoryBadgeClass(book.category)}`}>
   384→                              {book.category}
   385→                            </span>
   386→                          )}
   387→                        </TableCell>
   388→                        <TableCell className="text-center text-sm text-slate-300">{book.totalCopies}</TableCell>
   389→                        <TableCell className="text-center">
   390→                          <Badge
   391→                            className={`text-xs ${
   392→                              book.availableCopies > 0
   393→                                ? 'bg-emerald-500/20 text-emerald-400 border border-emerald-500/30'
   394→                                : 'bg-rose-500/20 text-rose-400 border border-rose-500/30'
   395→                            }`}
   396→                          >
   397→                            {book.availableCopies}
   398→                          </Badge>
   399→                        </TableCell>
   400→                        <TableCell>
   401→                          <div className="flex items-center justify-center gap-1">
   402→                            <Button
   403→                              variant="ghost"
   404→                              size="sm"
   405→                              onClick={() => handleEdit(book)}
   406→                              className="h-8 w-8 p-0 text-slate-400 hover:text-sky-400 hover:bg-slate-800"
   407→                            >
   408→                              <Pencil className="w-3.5 h-3.5" />
   409→                            </Button>
   410→                            <Button
   411→                              variant="ghost"
   412→                              size="sm"
   413→                              onClick={() => handleDelete(book)}
   414→                              className="h-8 w-8 p-0 text-slate-400 hover:text-rose-400 hover:bg-slate-800"
   415→                            >
   416→                              <Trash2 className="w-3.5 h-3.5" />
   417→                            </Button>
   418→                          </div>
   419→                        </TableCell>
   420→                      </TableRow>
   421→                    ))
   422→                  )}
   423→                </TableBody>
   424→              </Table>
   425→            </div>
   426→          </CardContent>
   427→        </Card>
   428→      ) : (
   429→        /* GRID VIEW */
   430→        <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 xl:grid-cols-5 gap-4">
   431→          {loading ? (
   432→            Array.from({ length: 8 }).map((_, i) => (
   433→              <Card key={i} className="bg-slate-900 border-slate-700">
   434→                <Skeleton className="w-full aspect-[3/4] bg-slate-800 rounded-t-xl" />
   435→                <CardContent className="p-3">
   436→                  <Skeleton className="h-4 w-3/4 bg-slate!-800 mb-2" />
   437→                  <Skeleton className="h-3 w-1/2 bg-slate-800" />
   438→                </CardContent>
   439→              </Card>
   440→            ))
   441→          ) : books.length === 0 ? (
   442→            <div className="col-span-full flex flex-col items-center justify-center py-12 text-slate-400">
   443→              <BookOpen className="w-12 h-12 mb-3 text-slate-600" />
   444→              <p>도서가 없습니다.</p>
   445→            </div>
   446→          ) : (
   447→            books.map((book) => (
   448→              <Card
   449→                key={book.id}
   450→                className="bg-slate-900 border-slate-700 text-white overflow-hidden group hover:border-slate-600 transition-colors duration-200"
   451→              >
   452→                {/* Cover image - top 60% */}
   453→                <BookCover coverUrl={book.coverUrl} title={book.title} size="grid" />
   454→                {/* Info */}
   455→                <CardContent className="p-3 space-y-2">
   456→                  <p className="font-medium text-sm text-white truncate" title={book.title}>
   457→                    {book.title}
   458→                  </p>
   459→                  <p className="text-xs text-slate-400 truncate">{book.author}</p>
   460→                  {book.category && (
   461→                    <span className={`inline-block text-[10px] px-1.5 py-0.5 rounded border ${getCategoryBadgeClass(book.category)}`}>

   469→                      대출가능: {book.availableCopies}

   471→                  </div>
   472→                  {/* Actions */}
   473→                  <div className="flex items-center gap-1 pt-1 border-t border-slate-700/50">

   475→                      variant="ghost"
   476→                      size="sm"
   477→                      onClick={() => handleEdit(book)}
   478→                      className="h-7 flex-1 text-slate-400 hover:text-sky-400 hover:bg-slate-800 text-xs"
   479→                    >
   480→                      <Pencil className="w-3 h-3 mr-1" />
   481→                      수정
   482→                    </Button>
   483→                    <Button
   484→                      variant="ghost"
   485→                      size="sm"
   486→                      onClick={() => handleDelete(book)}
   487→                      className="h-7 flex-1 text-slate-400 hover:text-rose-400 hover:bg-slate-800 text-xs"
   488→                    >
   489→                      <Trash2 className="w-3 h-3 mr-1" />
   490→                      삭제

   492→                  </div>
   493→                </CardContent>
   494→              </Card>

   496→          )}
   497→        </div>

   499→
   500→      {/* 도서 추가/수정 다이얼로그 - Dark styled */}
   501→      <Dialog open={editDialogOpen} onOpenChange={setEditDialogOpen}>
   502→        <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto bg-slate-900 border-slate-700 text-white">
   503→          <DialogHeader>
   504→            <DialogTitle className="text-white">{isAdding ? '도서 추가' : '도서 수정'}</DialogTitle>
   505→            <DialogDescription className="text-slate-400">
   506→              {isAdding ? '새로운 도서 정보를 입력하세요.' : '도서 정보를 수정하세요.'}
   507→            </DialogDescription>
   508→          </DialogHeader>
