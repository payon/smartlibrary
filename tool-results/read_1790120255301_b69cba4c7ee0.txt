   306→   306→export default function CardsSection() {
   307→   307→  // 상태
   308→   308→  const [cards, setCards] = useState<LibraryCard[]>([]);
   309→   309→  const [stats, setStats] = useState<CardStats>({ total: 0, pending: 0, approved: 0, issued: 0, rejected: 0 });

   312→   312→  const [statusFilter, setStatusFilter] = useState<string>('all');
   313→   313→  const [cardTypeFilter, setCardTypeFilter] = useState<string>('all');
   314→   314→
   315→   315→  // 다이얼로그
   316→   316→  const [detailDialogOpen, setDetailDialogOpen] = useState(false);
   317→   317→  const [issueDialogOpen, setIssueDialogOpen] = useState(false);
   318→   318→  const [rejectDialogOpen, setRejectDialogOpen] = useState(false);
   319→   319→  const [selectedCard, setSelectedCard] = useState<LibraryCard | null>(null);
   320→   320→  const [rejectReason, setRejectReason] = useState('');
   321→   321→  const [saving, setSaving] = useState(false);
   322→   322→
   323→   323→  // 직접 발급 폼
   324→   324→  const [issueForm, setIssueForm] = useState({
   325→   325→    applicantName: '',
   326→   326→    birthDate: '',
   327→   327→    phone: '',
   328→   328→    address: '',
   329→   329→    cardType: 'mobile',

   331→   331→
   332→   332→  // ==========================================================================
   333→   333→  // 데이터 로드
   334→   334→  // ==========================================================================
   335→   335→
   336→   336→  const fetchCards = useCallback(async (searchVal?: string, statusVal?: string, cardTypeVal?: string) => {
   337→   337→    try {
   338→   338→      setLoading(true);
   339→   339→      const params = new URLSearchParams();
   340→   340→      if (searchVal) params.set('search', searchVal);
   341→   341→      if (statusVal && statusVal !== 'all') params.set('status', statusVal);
   342→   342→      if (cardTypeVal && cardTypeVal !== 'all') params.set('cardType', cardTypeVal);
   343→   343→      const res = await fetch(`/api/admin/cards?${params.toString()}`);
   344→   344→      if (!res.ok) throw new Error();
   345→   345→      const json = await res.json();
   346→   346→      setCards(json.cards || []);
   347→   347→      if (json.stats) setStats(json.stats);
   348→   348→    } catch {
   349→   349→      toast.error('도서카드 목록을 불러오는데 실패했습니다.');
   350→   350→    } finally {
   351→   351→      setLoading(false);
   352→   352→    }
   353→   353→  }, []);
   354→   354→
   355→   355→  useEffect(() => {
   356→   356→    fetchCards();
   357→   357→  }, [fetchCards]);

   360→   360→  // 액션 핸들러

   363→   363→  const handleCardAction = async (cardId: string, action: string, extra?: Record<string, unknown>) => {
   364→   364→    setSaving(true);
   365→   365→    try {
   366→   366→      const res = await fetch(`/api/admin/cards/${cardId}`, {
   367→   367→        method: 'PUT',
   368→   368→        headers: { 'Content-Type': 'application/json' },
   369→   369→        body: JSON.stringify({ action, ...extra }),
   370→   370→      });
   371→   371→      if (!res.ok) {
   372→   372→        const data = await res.json();
   373→   373→        throw new Error(data.error || '작업 실패');
   374→   374→      }
   375→   375→      const data = await res.json();
   376→   376→      toast.success(data.message || '작업이 완료되었습니다.');
   377→   377→      await fetchCards(search, statusFilter, cardTypeFilter);
   378→   378→    } catch (err) {
   379→   379→      toast.error(err instanceof Error ? err.message : '작업에 실패했습니다.');
   380→   380→    } finally {
   381→   381→      setSaving(false);
   382→   382→    }
   383→   383→  };
   384→   384→
   385→   385→  const handleApprove = (card: LibraryCard) => handleCardAction(card.id, 'approve');
   386→   386→  const handleIssue = (card: LibraryCard) => handleCardAction(card.id, 'issue');
   387→   387→
   388→   388→  const handleReject = async () => {

   390→   390→    await handleCardAction(selectedCard.id, 'reject', { reason: rejectReason });
   391→   391→    setRejectDialogOpen(false);
   392→   392→    setRejectReason('');
   393→   393→    setSelectedCard(null);
   394→   394→  };
   395→   395→
   396→   396→  const handleDelete = async (card: LibraryCard) => {
   397→   397→    if (!confirm(`'${card.applicantName}'님의 도서카드 신청을 삭제하시겠습니까?`)) return;
   398→   398→    try {
   399→   399→      const res = await fetch(`/api/admin/cards/${card.id}`, { method: 'DELETE' });
   400→   400→      if (!res.ok) throw new Error();
   401→   401→      toast.success('도서카드 신청이 삭제되었습니다.');
   402→   402→      await fetchCards(search, statusFilter, cardTypeFilter);
   403→   403→    } catch {
   404→   404→      toast.error('삭제에 실패했습니다.');
   405→   405→    }
   406→   406→  };
   407→   407→
   408→   408→  const handleDirectIssue = async () => {
   409→   409→    const { applicantName, birthDate, phone } = issueForm;
   410→   410→    if (!applicantName || !birthDate || !phone) {
   411→   411→      toast.error('이름, 생년월일, 전화번호는 필수입니다.');
   412→   412→      return;
   413→   413→    }
   414→   414→    if (birthDate.length !== 8) {
   415→   415→      toast.error('생년월일은 8자리로 입력하세요.');
